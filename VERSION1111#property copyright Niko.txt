string StripBrokerSuffix(const string sym)
{
   string s = sym;
   StringToUpper(s);
   string suf[] = {".MICRO",".MINI",".RAW",".ECN",".PRO",".STD",".STP",
                   ".R",".S",".A",".M",".I",".C",".E",".P",".T",".X"};
   for(int i=0;i<ArraySize(suf);i++)
     {
      int n = StringLen(suf[i]);
      if(StringLen(s)>n && StringSubstr(s, StringLen(s)-n)==suf[i])
         return StringSubstr(sym, 0, StringLen(sym)-n);
     }
   return sym;
}

string ResolveBrokerSymbol(const string baseIn)
{
   string base = StripBrokerSuffix(baseIn);
   if(SymbolSelect(base,true) && SymbolInfoDouble(base,SYMBOL_BID)>0.0)
      return base;
   string bu = base;
   StringToUpper(bu);
   int total = SymbolsTotal(false);
   for(int i=0;i<total;i++)
     {
      string name = SymbolName(i,false);
      string nu = name;
      StringToUpper(nu);
      string ns = StripBrokerSuffix(nu);
      StringToUpper(ns);
      if(ns==bu || nu==bu)
        {
         SymbolSelect(name,true);
         if(SymbolInfoDouble(name,SYMBOL_BID)>0.0)
            return name;
        }
     }
   string trySuf[] = {".r",".s",".a",".m",".i",".c",".e",".pro",".raw",".ecn",".micro"};
   for(int j=0;j<ArraySize(trySuf);j++)
     {
      string t = base + trySuf[j];
      if(SymbolSelect(t,true) && SymbolInfoDouble(t,SYMBOL_BID)>0.0)
         return t;
     }
   return "";
}

int SnapLeverageTier(const double raw)
{
   if(raw<=0.0)   return 100;
   if(raw<50.0)   return 30;
   if(raw<150.0)  return 100;
   if(raw<250.0)  return 200;
   if(raw<350.0)  return 300;
   if(raw<450.0)  return 400;
   if(raw<750.0)  return 500;
   return 1000;
}

double DetectBrokerLeverage()
{
   long acc = AccountInfoInteger(ACCOUNT_LEVERAGE);
   int tier = SnapLeverageTier((double)acc);
   Print("Auto-detect leverage: broker 1:", acc, " → tier 1:", tier,
         " (input MaxLeverage=", DoubleToString(MaxLeverage,0), ")");
   return (double)tier;
}

double EffectiveLeverageCap()

}

bool NewBarTF(const ENUM_TIMEFRAMES tf, datetime &last)
{
   datetime currentBar = iTime(Symbol(), tf, 0);
   if(currentBar == 0)
      return false;
   if(currentBar != last)
     {
      last = currentBar;
      return true;
     }
   return false;
}

int ArraySearchString(string &arr[], string value) {
   string v = StripBrokerSuffix(value);
   StringToUpper(v);
   for (int i = 0; i < ArraySize(arr); i++) {
      if (arr[i] == value) return i;
      string a = StripBrokerSuffix(arr[i]);
      StringToUpper(a);
      if (a == v) return i;
   }
   return -1;
}

bool AiVoteAligned(const string voteMode, const double conf)
{
   string m = voteMode;
   StringToLower(m);
   bool multi = (StringFind(m, "multi") >= 0 || StringFind(m, "consensus") >= 0);
   if(!multi)
      return false;
   if(conf < MinAIConfidence || conf > MaxAIConfidence) return false;
   if(conf >= 0.9999 || conf >= 1.0) return false;
   return true;
}

double CalculateLotSize(string pair, double &slPrice, double openPrice, ENUM_ORDER_TYPE orderType,
                        const double aiConf = 0.0, const string voteMode = "") {
   double balance;
   if (AccountSize == Acct_Custom && CustomAccountSize > 0.0)
      balance = CustomAccountSize;
   else if (AccountSize != Acct_Custom)
      balance = (double)AccountSize;
   else
      balance = AccountInfoDouble(ACCOUNT_BALANCE);

   double riskPct = RiskPercent;
   bool aligned = AiVoteAligned(voteMode, aiConf);
   // At 100% risk the lot is already "all that margin allows". Do not 2x/3x that.
   if(aligned && RiskPercent + 1e-12 < 100.0)
     {
      if(aiConf >= 0.88)
         riskPct = RiskPercent * AlignStrongMult;
      else
         riskPct = RiskPercent * AlignRiskMult;
     }
   if(riskPct > 100.0)
      riskPct = 100.0;
   double riskAmount = balance * (riskPct / 100.0);
   double pipValue   = SymbolInfoDouble(pair, SYMBOL_TRADE_TICK_VALUE);
   double point      = SymbolInfoDouble(pair, SYMBOL_POINT);
   int    digits     = (int)SymbolInfoInteger(pair, SYMBOL_DIGITS);
   double brokerMin  = SymbolInfoDouble(pair, SYMBOL_VOLUME_MIN);
   double maxLot     = SymbolInfoDouble(pair, SYMBOL_VOLUME_MAX);
   double stepLot    = SymbolInfoDouble(pair, SYMBOL_VOLUME_STEP);

   // No hard 0.01/0.1 lock — use broker SYMBOL_VOLUME_MIN / STEP only
   if (stepLot <= 0.0) stepLot = brokerMin;
   if (stepLot <= 0.0) stepLot = 0.01;
   double hardMin = brokerMin;
   if (hardMin <= 0.0) hardMin = stepLot;
   int volDigits = 2;
   if (stepLot < 0.01 - 1e-12) volDigits = 3;
   else if (stepLot >= 1.0 - 1e-12) volDigits = 0;
   hardMin = NormalizeDouble(hardMin, volDigits);

   if (point == 0 || pipValue == 0) return 0.0;
   if (balance <= 0.0) {
      Print("LOT SKIP: ", pair, " balance/CustomAccountSize is $0 — set CustomAccountSize=50 for Vantage micro");
      return 0.0;
   }
   double slDistance = MathAbs(openPrice - slPrice) / point;
   if (slDistance == 0) return 0.0;

   // Risk-based calculation (optional HFT SymbolInfo position size)
   double lotSize = riskAmount / (slDistance * pipValue);
   if(UsePositionSizeMqh)
     {
      double hftLots = HFT_PositionSize(pair, riskAmount, slDistance, maxLot);
      if(hftLots > 0.0)
         lotSize = hftLots;
     }
   // Snap to step grid from hardMin upward
   int steps = (int)MathFloor((lotSize - hardMin) / stepLot + 1e-12);
   lotSize = hardMin + MathMax(0, steps) * stepLot;
   lotSize = NormalizeDouble(MathMax(hardMin, MathMin(maxLot, lotSize)), volDigits);

   // MaxTradeRiskPct >= 100 → full risk, no skip/SL-clamp from risk cap
   double maxAcceptableRisk = (MaxTradeRiskPct >= 100.0)
                              ? (balance * 1000.0)   // effectively unlimited
                              : balance * (MaxTradeRiskPct / 100.0);
   double minLotRisk        = hardMin * slDistance * pipValue;

   if (MaxTradeRiskPct < 100.0 && lotSize <= hardMin && minLotRisk > maxAcceptableRisk) {
      double affordableDist = maxAcceptableRisk / (hardMin * pipValue);
      if (affordableDist < 1) {
         Print("SKIP: ", pair, " un-tradeable (min lot ", DoubleToString(hardMin, volDigits),
               " 1-pt risk $", DoubleToString(hardMin * pipValue, 2), " > max $",
               DoubleToString(maxAcceptableRisk, 2), " on $", DoubleToString(balance, 2), ")");
         return 0.0;
      }
      double oldSL = slPrice;
      if (orderType == ORDER_TYPE_BUY)
         slPrice = NormalizeDouble(openPrice - affordableDist * point, digits);
      else
         slPrice = NormalizeDouble(openPrice + affordableDist * point, digits);
      lotSize = hardMin;
      Print("SL CLAMPED: ", pair, " SL ", DoubleToString(oldSL, digits),
            " -> ", DoubleToString(slPrice, digits),
            " (risk $", DoubleToString(minLotRisk, 2), " -> $",
            DoubleToString(maxAcceptableRisk, 2), " on $", DoubleToString(balance, 2), ")");
   }
   else if (lotSize <= hardMin && minLotRisk > riskAmount && balance > 0.0) {
      Print("MIN-LOT: ", pair, " ", DoubleToString(hardMin, volDigits), " lot risk $",
            DoubleToString(minLotRisk, 2),
            " (", DoubleToString(minLotRisk / balance * 100.0, 1), "% vs intended ",
            DoubleToString(RiskPercent, 1), "%)");
   }

   // Min-lot only when NOT an aligned multi-model vote
   if(HolyMinLotFirst && !aligned)
      lotSize = hardMin;
   lotSize = HolySafeLots(pair, orderType, openPrice, lotSize, hardMin, stepLot, maxLot, volDigits);
   if(lotSize <= 0.0)
      return 0.0;

   Print("Symbol: ", pair,
         " Min: ", DoubleToString(hardMin, volDigits),
         " Max: ", DoubleToString(maxLot, volDigits),
         " Step: ", DoubleToString(stepLot, volDigits),
         " Lot: ", DoubleToString(lotSize, volDigits),
         " Bal: $", DoubleToString(balance, 2),
         " risk%=", DoubleToString(riskPct, 2),
         " aligned=", (aligned ? "yes" : "no"),
         " vote=", voteMode);
   return lotSize;
}

bool HolyConfInBand(const double conf)
{
   if(conf < MinAIConfidence) return false;
   if(conf > MaxAIConfidence) return false;
   if(conf >= 0.9999) return false;
   if(conf >= 1.0) return false;
   return true;
}

double HolySafeLots(const string pair, const ENUM_ORDER_TYPE orderType, const double openPrice,
                    double lotSize, const double hardMin, const double stepLot,
                    const double maxLot, const int volDigits)
{
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   if(equity <= 0.0) equity = AccountInfoDouble(ACCOUNT_BALANCE);
   double freeMargin = AccountInfoDouble(ACCOUNT_MARGIN_FREE);
   if(equity < 10.0 || freeMargin <= 0.0)
     {
      Print("HOLY SKIP: ", pair, " insufficient funds/margin eq=$",
            DoubleToString(equity, 2), " free=$", DoubleToString(freeMargin, 2));
      return 0.0;
     }
   // Stop-out cushion only. If reserve would skip the trade, fall back to 80% of free.
   double reserve = equity * (HolyMarginReservePct / 100.0);
   double usable = freeMargin - reserve;
   if(usable < hardMin * 0.0 + 1e-8)
      usable = freeMargin * 0.80;
   double maxUse = freeMargin * (HolyMaxMarginUsePct / 100.0);
   if(maxUse <= 0.0)
      maxUse = freeMargin * 0.80;
   if(usable > maxUse)
      usable = maxUse;
   if(usable <= 0.0)
      usable = freeMargin * 0.80;

   if(lotSize < hardMin) lotSize = hardMin;
   // Shrink to what margin allows. Never skip a trade that still fits min lot.
   for(int guard = 0; guard < 80; guard++)
     {
      double marginRequired = 0.0;
      if(!OrderCalcMargin(orderType, pair, lotSize, openPrice, marginRequired))
        {
         Print("HOLY SKIP: margin calc failed ", pair);
         return 0.0;
        }
      if(marginRequired > usable)
        {
         if(lotSize <= hardMin + 1e-12)
           {
            // Last try: min lot against raw free margin (keep 20% so we are not stopped out)
            double emergency = freeMargin * 0.80;
            if(marginRequired <= emergency && emergency > 0.0)
              {
               Print("MARGIN FIT: ", pair, " min lot on 80% free $", DoubleToString(emergency, 2));
               break;
              }
            Print("HOLY SKIP: ", pair, " min lot needs $", DoubleToString(marginRequired, 2),
                  " free=$", DoubleToString(freeMargin, 2));
            return 0.0;
           }
         lotSize = NormalizeDouble(lotSize - stepLot, volDigits);
         if(lotSize < hardMin) lotSize = hardMin;
         continue;
        }
      double lev = CalculateTrueLeverage(lotSize, pair);
      double cap = HolyLeverageCeiling;
      if(cap < 5.0) cap = 5.0;
      if(lev > cap)
        {
         if(lotSize <= hardMin + 1e-12)
           {
            Print("LEV NOTE: ", pair, " min lot 1:", DoubleToString(lev, 1),
                  " over ceiling 1:", DoubleToString(cap, 1), " — still taking min lot");
            break;
           }
         lotSize = NormalizeDouble(lotSize - stepLot, volDigits);
         if(lotSize < hardMin) lotSize = hardMin;
         continue;
        }
      break;
     }
   lotSize = NormalizeDouble(MathMax(hardMin, MathMin(maxLot, lotSize)), volDigits);
   return lotSize;
}

bool HolyRegimeVolumeFlip(const string sym, const long posType)
{
   int idx = ArraySearchString(dynamicPairList, sym);
   if(idx < 0) return false;
   int dir = (posType == POSITION_TYPE_BUY) ? 1 : -1;
   if(UseRegimeHelpers && Saturn_InRange(idx, ArraySize(g_regPack)))
     {
      if(g_regPack[idx].regime_id == 2 && dir > 0) return true;  // downtrend vs long
      if(g_regPack[idx].regime_id == 1 && dir < 0) return true;  // uptrend vs short
     }
   if(UseVolumeHelpers && Saturn_InRange(idx, ArraySize(g_volPack)))
     {
      if(g_volPack[idx].volume_surge >= 1.0 && g_volPack[idx].volume_trend == (double)(-dir))
         return true;
     }
   return false;
}

double CalculateTrueLeverage(double additionalLots, string pair) {
   // Notional / equity (MT5). Floor equity at $10 so micro accounts still size.
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   if (equity <= 0.0) equity = AccountInfoDouble(ACCOUNT_BALANCE);
   if (equity < 10.0 && g_initialBalance >= 10.0) equity = g_initialBalance;
   if (equity <= 0.0) return 0.0;
   double notional = 0.0;
   for(int i=0;i<PositionsTotal();i++)
     {
      ulong t = PositionGetTicket(i);
      if(!PositionSelectByTicket(t)) continue;
      string ps = PositionGetString(POSITION_SYMBOL);
      double vol = PositionGetDouble(POSITION_VOLUME);
      double px  = PositionGetDouble(POSITION_PRICE_CURRENT);
      double cs  = SymbolInfoDouble(ps, SYMBOL_TRADE_CONTRACT_SIZE);
      if(cs<=0.0) cs = 100000.0;
      if(px<=0.0) px = SymbolInfoDouble(ps, SYMBOL_BID);
      notional += vol * cs * px;
     }
   if(!symbolInfo.Name(pair)) return notional / equity;
   double cs2 = symbolInfo.ContractSize();
   double bid = SymbolInfoDouble(pair, SYMBOL_BID);
   if(cs2<=0.0) cs2 = 100000.0;
   notional += additionalLots * cs2 * bid;
   return notional / equity;
}

double GetDynamicSlippage(string sym, int idx) {
   if (idx < 0 || idx >= totalPairs || ArraySize(cachedATR) <= idx) {
      Print("Invalid index ", idx, " for slippage in ", sym);
      return MaxSlippagePips;

   double atrPips = cachedATR[idx] / point;
   double slippage = MathMin(MaxSlippagePips * 10, atrPips * 2.5);
   Print("Slippage for ", sym, ": ", slippage, " points (ATR: ", atrPips, ")");
   return slippage;
}

double GetAdjustedCarryRate(string pair) {
   double swapLong = SymbolInfoDouble(pair, SYMBOL_SWAP_LONG);
   double swapShort = SymbolInfoDouble(pair, SYMBOL_SWAP_SHORT);
   MqlDateTime timeStruct;
   TimeToStruct(TimeCurrent(), timeStruct);
   if (timeStruct.day_of_week == 3) { // Triple swaps on Wednesday
      swapLong *= 3.0;
      swapShort *= 3.0;
   }
   return (swapLong > swapShort) ? swapLong : swapShort;
}

void CalculateVWAP(int idx, string symbol) {
   if (idx < 0 || idx >= totalPairs || ArraySize(cachedVWAP) <= idx) return;
   double high[], low[], close[];
   long volumes[];
   ArraySetAsSeries(high, true); ArraySetAsSeries(low, true);
   ArraySetAsSeries(close, true); ArraySetAsSeries(volumes, true);
   if (CopyHigh(symbol, PERIOD_M15, 0, VWAP_Period, high) < VWAP_Period ||
       CopyLow(symbol, PERIOD_M15, 0, VWAP_Period, low) < VWAP_Period ||
       CopyClose(symbol, PERIOD_M15, 0, VWAP_Period, close) < VWAP_Period ||
       CopyTickVolume(symbol, PERIOD_M15, 0, VWAP_Period, volumes) < VWAP_Period) return;
   double sumPriceVolume = 0.0, sumVolume = 0.0;
   double typicalPrices[]; ArrayResize(typicalPrices, VWAP_Period);
   for (int i = 0; i < VWAP_Period; i++) {
      double typicalPrice = (high[i] + low[i] + close[i]) / 3.0;
      sumPriceVolume += typicalPrice * (double)volumes[i];
      sumVolume += (double)volumes[i];
      typicalPrices[i] = typicalPrice;
   }
   if (sumVolume > 0) {
      cachedVWAP[idx] = sumPriceVolume / sumVolume;
      double sumVariance = 0.0;
      for (int i = 0; i < VWAP_Period; i++)
         sumVariance += MathPow(typicalPrices[i] - cachedVWAP[idx], 2) * (double)volumes[i];
      double std_dev = MathSqrt(sumVariance / sumVolume);
      cachedVWAPUpper[idx] = cachedVWAP[idx] + VWAP_Bands_Multiplier * std_dev;
      cachedVWAPLower[idx] = cachedVWAP[idx] - VWAP_Bands_Multiplier * std_dev;
   }
}

void ValidateDynamicPairList() {
   for (int i = ArraySize(dynamicPairList) - 1; i >= 0; i--) {
      string resolved = ResolveBrokerSymbol(dynamicPairList[i]);
      if (resolved == "") {
         Print("Removing unavailable symbol: ", dynamicPairList[i]);
         ArrayRemove(dynamicPairList, i);
      } else {
         if (resolved != dynamicPairList[i])
            Print("Symbol mapped ", dynamicPairList[i], " → ", resolved,
                  " (base=", StripBrokerSuffix(resolved), ")");
         dynamicPairList[i] = resolved;
      }
   }
   totalPairs = ArraySize(dynamicPairList);
   if (totalPairs == 0) {
      Print("No valid symbols in dynamicPairList. EA will not trade.");
      tradingEnabled = false;
   } else {
      Print("Active symbols (", totalPairs, "): ", ArrayToStringCustom(dynamicPairList));
   }
}

void ArrayRemove(string &arr[], int index) {
   int size = ArraySize(arr);
   if (index < 0 || index >= size) return;
   // Must shrink array — ArrayCopy alone leaves old size (caused duplicate XRPUSD rows)
   for (int i = index; i < size - 1; i++)
      arr[i] = arr[i + 1];
   ArrayResize(arr, size - 1);
}

// ── Per-symbol trading-hours check ───────────────────────────────────────────
// Forex:  allowed 07:45-12:15 UTC, 13:45-15:15 UTC, 16:45-20:15 UTC
//         (night 21:00-06:00 + ±45 min session-transition buffers excluded)
// Crypto: allowed all weekday hours except ±45 min session transitions
//         London-open(07:00), NY-open(13:00), London-close(16:00), NY-close(21:00)
// Sunday:  always blocked; Friday ≥ 21:00 UTC: blocked
bool IsTradingAllowedForSymbol(string sym) {
   if (BypassSessionFilter) return true;    // testing override
   MqlDateTime dt; TimeToStruct(TimeGMT(), dt);
   int h = dt.hour, m = dt.min;
   int tod = h * 60 + m;                    // minutes-of-day in UTC

   // Weekend guards
   if (dt.day_of_week == 0) return false;   // Sunday always off
   if (dt.day_of_week == 5 && h >= 21) return false; // Friday close

   bool isCrypto = (StringFind(sym,"BTC")>=0 || StringFind(sym,"ETH")>=0 ||
                    StringFind(sym,"XRP")>=0 || StringFind(sym,"LTC")>=0);

   // Forex-only: hard night block 21:00-07:00 UTC
   if (!isCrypto && (h >= 21 || h < 7)) return false;

   // ±45 min buffers around session open/close times (UTC minutes-of-day)
   // London open 07:00 = 420, NY open 13:00 = 780,
   // London close 16:00 = 960, NY close 21:00 = 1260
   int boundaries[4] = {420, 780, 960, 1260};
   int buf = 45;
   for (int i = 0; i < 4; i++) {
      int diff = tod - boundaries[i];
      if (diff < -720) diff += 1440;
      if (diff >  720) diff -= 1440;
      if (MathAbs(diff) <= buf) return false;  // inside buffer window
   }
   return true;
}

void UpdateGARCHVolatility() {
   for (int i = 0; i < totalPairs; i++) {
      double price = SymbolInfoDouble(dynamicPairList[i], SYMBOL_BID);
      if (lastPrice[i] > 0) {
         double ret = MathLog(price / lastPrice[i]);
         double garch_var = garchOmega + garchAlpha * ret * ret + garchBeta * garchVolatility[i] * garchVolatility[i];
         garchVolatility[i] = MathSqrt(MathMax(garch_var, 1e-8));  // MathMax prevents zero/negative variance
      }
      lastPrice[i] = price;
   }
}

void MultiPartialExit() {
   for (int i = PositionsTotal() - 1; i >= 0; i--) {
      ulong ticket = PositionGetTicket(i);
      if (!PositionSelectByTicket(ticket)) continue;

      string sym   = PositionGetString(POSITION_SYMBOL);
      int    type  = (int)PositionGetInteger(POSITION_TYPE);
      double op    = PositionGetDouble(POSITION_PRICE_OPEN);
      double vol   = PositionGetDouble(POSITION_VOLUME);
      double sl    = PositionGetDouble(POSITION_SL);
      double tp    = PositionGetDouble(POSITION_TP);
      double pts   = SymbolInfoDouble(sym, SYMBOL_POINT);
      int    digs  = (int)SymbolInfoInteger(sym, SYMBOL_DIGITS);
      double currentPrice = (type == POSITION_TYPE_BUY)
                            ? SymbolInfoDouble(sym, SYMBOL_BID)
                            : SymbolInfoDouble(sym, SYMBOL_ASK);

      // Look up ATR for this symbol
      int idx = ArraySearchString(dynamicPairList, sym);
      double atr = (idx >= 0 && cachedATR[idx] > 0) ? cachedATR[idx] : 0;

      double profitPips = (type == POSITION_TYPE_BUY) ? (currentPrice - op) / pts
                                                      : (op - currentPrice) / pts;

      // ── 1. PARTIAL CLOSE at TP1 ──────────────────────────────────
      // Close PartialClosePercent (default 50%) of position at TP1 level,
      // then move SL to breakeven + offset to cover slippage/fees.
      if (profitPips >= PartialExitPips && profitPips < TP2Pips && vol > 0) {
         double stepLot = SymbolInfoDouble(sym, SYMBOL_VOLUME_STEP);
         double closeVolume = MathFloor(vol * PartialClosePercent / 100.0 / stepLot) * stepLot;
         if (closeVolume < stepLot) closeVolume = stepLot;
         if (closeVolume > vol) closeVolume = vol;
         if (closeVolume >= SymbolInfoDouble(sym, SYMBOL_VOLUME_MIN)) {
            if (trade.PositionClosePartial(ticket, closeVolume)) {
               // Move SL to breakeven + offset (covers spread/fees/slippage)
               double beSL = (type == POSITION_TYPE_BUY)
                             ? NormalizeDouble(op + BreakevenOffsetPips * pts, digs)
                             : NormalizeDouble(op - BreakevenOffsetPips * pts, digs);
               // Only tighten SL, never widen
               bool shouldMove = (type == POSITION_TYPE_BUY) ? (beSL > sl) : (beSL < sl || sl == 0);
               if (shouldMove) trade.PositionModify(ticket, beSL, tp);
               Print("TP1 partial close: ", DoubleToString(closeVolume, 2), " lots of #", ticket,
                     ", SL→BE+", DoubleToString(BreakevenOffsetPips, 1), " pips");
            }
         }
      }
      // TP2: close half of remaining position
      else if (profitPips >= TP2Pips && vol > 0) {
         double stepLot = SymbolInfoDouble(sym, SYMBOL_VOLUME_STEP);
         double closeVolume = MathFloor(vol / 2.0 / stepLot) * stepLot;
         if (closeVolume < stepLot) closeVolume = stepLot;
         if (closeVolume > vol) closeVolume = vol;
         if (closeVolume >= SymbolInfoDouble(sym, SYMBOL_VOLUME_MIN)) {
            if (trade.PositionClosePartial(ticket, closeVolume)) {
               Print("TP2 partial close: ", DoubleToString(closeVolume, 2), " lots of #", ticket);
            }
         }
      }

      // ── 2. BREAKEVEN SL ──────────────────────────────────────────
      // When price moves ATR * BreakevenTriggerATRMult in our favor,
      // lock in profit by moving SL to entry + offset (covers fees/slippage).
      if (atr > 0) {
         double beThreshold = atr * BreakevenTriggerATRMult;
         double profitDistance = (type == POSITION_TYPE_BUY) ? (currentPrice - op) : (op - currentPrice);

         if (profitDistance >= beThreshold) {
            double beSL = (type == POSITION_TYPE_BUY)
                          ? NormalizeDouble(op + BreakevenOffsetPips * pts, digs)
                          : NormalizeDouble(op - BreakevenOffsetPips * pts, digs);
            bool shouldMove = (type == POSITION_TYPE_BUY) ? (beSL > sl) : (beSL < sl || sl == 0);
            if (shouldMove) {
               if (trade.PositionModify(ticket, beSL, tp))
                  Print("Breakeven SL: #", ticket, " SL→", DoubleToString(beSL, digs));
            }
         }
      }

      // ── 3. TRAILING STOP ─────────────────────────────────────────
      // Once in profit, trail SL at ATR * TrailingStopATRMult behind price.
      // Only tightens — never moves SL further from price.
      if (atr > 0) {
         double trailDist = atr * TrailingStopATRMult;
         double profitDistance = (type == POSITION_TYPE_BUY) ? (currentPrice - op) : (op - currentPrice);

         // Only trail once we're at least trailDist in profit (avoids trailing on noise)
         if (profitDistance > trailDist) {
            double newSL;
            if (type == POSITION_TYPE_BUY) {
               newSL = NormalizeDouble(currentPrice - trailDist, digs);
               // Only move SL up, never down
               if (newSL > sl) {
                  if (trade.PositionModify(ticket, newSL, tp))
                     Print("Trailing SL: #", ticket, " SL→", DoubleToString(newSL, digs),
                           " (price=", DoubleToString(currentPrice, digs), ")");
               }
            } else {
               newSL = NormalizeDouble(currentPrice + trailDist, digs);
               // Only move SL down, never up
               if (newSL < sl || sl == 0) {
                  if (trade.PositionModify(ticket, newSL, tp))
                     Print("Trailing SL: #", ticket, " SL→", DoubleToString(newSL, digs),
                           " (price=", DoubleToString(currentPrice, digs), ")");
               }
            }
         }
      }
   }
}

bool HasMicroBreakoutSignal(string pair, int idx, FXJEFE_CandidateTrade &c) {
   if (!UseMicroBreakout) return false;
   double currentPrice = SymbolInfoDouble(pair, SYMBOL_BID);
   double filter = 0.3 * cachedATR[idx];
   MqlDateTime timeStruct; TimeToStruct(TimeCurrent(), timeStruct);
   int hour = timeStruct.hour;
   if (hour >= 8 && hour < 17) {
      if (tokyoHigh[idx] == 0 || tokyoLow[idx] == 0) return false;
      if (currentPrice > tokyoHigh[idx] + filter) {
         c.pair = pair; c.orderType = ORDER_TYPE_BUY; c.openPrice = currentPrice;
         c.stopLoss = tokyoLow[idx]; c.takeProfit = currentPrice + (currentPrice - tokyoLow[idx]) * 2;
         c.strategyName = "MicroBreakout"; c.momentumScore = GetMomentumScore(pair, idx);
         c.lotSize = CalculateLotSize(pair, c.stopLoss, c.openPrice, c.orderType);
         return true;
      } else if (currentPrice < tokyoLow[idx] - filter) {
         c.pair = pair; c.orderType = ORDER_TYPE_SELL; c.openPrice = currentPrice;
         c.stopLoss = tokyoHigh[idx]; c.takeProfit = currentPrice - (tokyoHigh[idx] - currentPrice) * 2;
         c.strategyName = "MicroBreakout"; c.momentumScore = GetMomentumScore(pair, idx);
         c.lotSize = CalculateLotSize(pair, c.stopLoss, c.openPrice, c.orderType);
         return true;
      }
   }
   return false;
}

      return true;
   } else if (currentPrice > cachedBBUpper[idx] && currentPrice > cachedVWAP[idx]) {
      c.pair = pair; c.orderType = ORDER_TYPE_SELL; c.openPrice = currentPrice;
      c.stopLoss = currentPrice + 2 * cachedATR[idx]; c.takeProfit = cachedBBLower[idx];
      c.strategyName = "StatArbitrage"; c.momentumScore = GetMomentumScore(pair, idx);
      c.lotSize = CalculateLotSize(pair, c.stopLoss, c.openPrice, c.orderType);
      return true;
   }
   return false;
}

bool HasCarryTradeSignal(string pair, int idx, FXJEFE_CandidateTrade &c) {
   if (!UseCarryTrade) return false;
   double swapLong = SymbolInfoDouble(pair, SYMBOL_SWAP_LONG);
   double swapShort = SymbolInfoDouble(pair, SYMBOL_SWAP_SHORT);
   double currentPrice = SymbolInfoDouble(pair, SYMBOL_BID);
   double carryRate = GetAdjustedCarryRate(pair);
   if (swapLong > 0 && currentPrice > cachedEMASlow[idx] && carryRate > 0.1) {
      c.pair = pair; c.orderType = ORDER_TYPE_BUY; c.openPrice = currentPrice;
      c.stopLoss = cachedEMASlow[idx] - cachedATR[idx]; c.takeProfit = currentPrice + 2 * cachedATR[idx];
      c.strategyName = "CarryTrade"; c.momentumScore = GetMomentumScore(pair, idx);
      c.lotSize = CalculateLotSize(pair, c.stopLoss, c.openPrice, c.orderType);
      return true;
   } else if (swapShort > 0 && currentPrice < cachedEMASlow[idx] && carryRate > 0.1) {
      c.pair = pair; c.orderType = ORDER_TYPE_SELL; c.openPrice = currentPrice;
      c.stopLoss = cachedEMASlow[idx] + cachedATR[idx]; c.takeProfit = currentPrice - 2 * cachedATR[idx];
      c.strategyName = "CarryTrade"; c.momentumScore = GetMomentumScore(pair, idx);
      c.lotSize = CalculateLotSize(pair, c.stopLoss, c.openPrice, c.orderType);
      return true;
   }
   return false;
}

//+------------------------------------------------------------------+
//| FetchSentiment — GET /predict/sentiment?symbol=BASE             |
//+------------------------------------------------------------------+
double FetchSentiment(string symbol)
{
   string base = StripBrokerSuffix(symbol);
   string url = AI_SENTIMENT_URL + "?symbol=" + base;
   char dummy[], result[];
   string result_headers;
   int res = WebRequest("GET", url, "", 5000, dummy, result, result_headers);
   if(res == 200)
   {
      string resp = CharArrayToString(result, 0, -1, CP_UTF8);
      int pos = StringFind(resp, "\"sentiment\":");
      if(pos >= 0)
      {
         string val = StringSubstr(resp, pos + 13);
         StringReplace(val, "}", "");
         StringTrimRight(val);
         return StringToDouble(val);
      }
   }
   return 0.0;
}

//+------------------------------------------------------------------+
//| ComputeExtendedFeatures — updates all M1 model feature caches    |
//| Called once per M15 bar for each symbol                         |
//+------------------------------------------------------------------+
void ComputeExtendedFeatures(int idx, string sym)
{
   if(idx < 0 || idx >= totalPairs) return;

   // ROC
   { double p[]; ArraySetAsSeries(p, true);
     if(CopyClose(sym, PERIOD_M1, 0, 15, p) >= 15 && p[14] != 0)
        cachedROC[idx] = ((p[0] - p[14]) / p[14]) * 100.0;
   }
   // CCI (M1)
   { double b[]; ArraySetAsSeries(b, true);
     int h = iCCI(sym, PERIOD_M1, 14, PRICE_TYPICAL);
     if(CopyBuffer(h, 0, 0, 1, b) > 0) cachedCCI[idx] = b[0];
     IndicatorRelease(h);
   }
   // Williams %R (M1)
   { double b[]; ArraySetAsSeries(b, true);
     int h = iWPR(sym, PERIOD_M1, 14);
     if(CopyBuffer(h, 0, 0, 1, b) > 0) cachedWilliams[idx] = b[0];
     IndicatorRelease(h);
   }
   // Momentum
   { double p[]; ArraySetAsSeries(p, true);
     if(CopyClose(sym, PERIOD_M1, 0, 15, p) >= 15)
        cachedMomentum[idx] = p[0] - p[14];
   }
   // Realized Volatility (M1 log returns)
   { double closes[]; ArraySetAsSeries(closes, true);
     if(CopyClose(sym, PERIOD_M1, 0, 15, closes) >= 15)
     {
        double returns[]; ArrayResize(returns, 14);
        bool ok = true;
        for(int j = 0; j < 14; j++) {
           if(closes[j+1] <= 0) { ok = false; break; }
           returns[j] = MathLog(closes[j] / closes[j+1]);
        }
        if(ok) {
           // Manual std-dev (replaces MathStandardDeviation — no Math\Stat\Math.mqh needed)
           double _mean = 0.0;
           for(int _k = 0; _k < 14; _k++) _mean += returns[_k];
           _mean /= 14.0;
           double _var = 0.0;
           for(int _k = 0; _k < 14; _k++) _var += (returns[_k] - _mean) * (returns[_k] - _mean);
           _var /= 14.0;
           cachedRealizedVol[idx] = MathSqrt(_var) * MathSqrt(252.0 * 60.0);
        }
     }
   }
   // Chaikin Volatility (M1)
   { double h1[], l1[], h2[], l2[];
     ArraySetAsSeries(h1,true); ArraySetAsSeries(l1,true);
     ArraySetAsSeries(h2,true); ArraySetAsSeries(l2,true);
     if(CopyHigh(sym, PERIOD_M1, 0,  14, h1) >= 14 && CopyLow(sym, PERIOD_M1, 0,  14, l1) >= 14 &&
        CopyHigh(sym, PERIOD_M1, 14, 14, h2) >= 14 && CopyLow(sym, PERIOD_M1, 14, 14, l2) >= 14)
     {
        double range = 0.0, prev_range = 0.0;
        for(int j = 0; j < 14; j++) { range += h1[j]-l1[j]; prev_range += h2[j]-l2[j]; }
        if(prev_range > 0) cachedChaikinVol[idx] = ((range - prev_range) / prev_range) * 100.0;
     }
   }
   // RVI (M1)
   { double close[], open[];
     ArraySetAsSeries(close, true); ArraySetAsSeries(open, true);
     if(CopyClose(sym, PERIOD_M1, 0, 14, close) >= 14 && CopyOpen(sym, PERIOD_M1, 0, 14, open) >= 14)
     {
        double co_sum = 0.0, sq_sum = 0.0;
        for(int j = 0; j < 14; j++) { co_sum += close[j]-open[j]; sq_sum += MathPow(close[j],2); }
        double denom = MathSqrt(sq_sum / 14.0);
        if(denom > 0) cachedRVI[idx] = (co_sum / 14.0) / denom;
     }
   }
   // OBV (M1, last bar delta)
   { double closes[]; long tvol[];
     ArraySetAsSeries(closes, true); ArraySetAsSeries(tvol, true);
     if(CopyClose(sym, PERIOD_M1, 0, 2, closes) >= 2 && CopyTickVolume(sym, PERIOD_M1, 0, 1, tvol) >= 1)
     {
        double prev = cachedOBV[idx];
        if(closes[0] > closes[1])      cachedOBV[idx] = prev + (double)tvol[0];
        else if(closes[0] < closes[1]) cachedOBV[idx] = prev - (double)tvol[0];
     }
   }
   // Volume Delta (M1)
   { long tvol[]; ArraySetAsSeries(tvol, true);
     if(CopyTickVolume(sym, PERIOD_M1, 0, 1, tvol) >= 1)
     {
        double pc = iClose(sym, PERIOD_M1, 0) - iOpen(sym, PERIOD_M1, 0);
        cachedVolumeDelta[idx] = pc >= 0 ? (double)tvol[0] : -(double)tvol[0];
     }
   }
   // AD Line (M1, running sum)
   { double c = iClose(sym,PERIOD_M1,0), h = iHigh(sym,PERIOD_M1,0), l = iLow(sym,PERIOD_M1,0);
     long tvol[]; ArraySetAsSeries(tvol, true);
     if(CopyTickVolume(sym, PERIOD_M1, 0, 1, tvol) >= 1)
     {
        double mfm = (h==l) ? 0.0 : ((c-l)-(h-c))/(h-l);
        cachedADLine[idx] += mfm * (double)tvol[0];
     }
   }
   // Volume Oscillator (M1)
   { long tvol[]; ArraySetAsSeries(tvol, true);
     if(CopyTickVolume(sym, PERIOD_M1, 0, 14, tvol) >= 14)
     {
        double sma5 = 0.0, sma14 = 0.0;
        for(int j = 0; j < 5;  j++) sma5  += (double)tvol[j];
        for(int j = 0; j < 14; j++) sma14 += (double)tvol[j];
        sma5 /= 5.0; sma14 /= 14.0;
        if(sma14 > 0) cachedVolOsc[idx] = (sma5 - sma14) / sma14;
     }
   }
   // Supertrend (M1)
   { double high[], low[], close_arr[]; double atr_buf[];
     ArraySetAsSeries(high, true); ArraySetAsSeries(low, true);
     ArraySetAsSeries(close_arr, true); ArraySetAsSeries(atr_buf, true);
     int h = iATR(sym, PERIOD_M1, 14);
     if(CopyHigh(sym,  PERIOD_M1, 0, 15, high)      >= 15 &&
        CopyLow(sym,   PERIOD_M1, 0, 15, low)       >= 15 &&
        CopyClose(sym, PERIOD_M1, 0, 15, close_arr) >= 15 &&
        CopyBuffer(h, 0, 0, 1, atr_buf) > 0)
     {
        double hl2   = (high[0] + low[0]) / 2.0;
        double upper = hl2 + 3.0 * atr_buf[0];
        double lower = hl2 - 3.0 * atr_buf[0];
        cachedSupertrend[idx] = close_arr[1] > upper ? 1.0 : close_arr[1] < lower ? -1.0 : 0.0;
     }
     IndicatorRelease(h);
   }
   // HMA (M1, period=9)
   { double prices[]; ArraySetAsSeries(prices, true);
     if(CopyClose(sym, PERIOD_M1, 0, 18, prices) >= 18)
     {
        double wma1 = 0.0, wma2 = 0.0;
        for(int j = 0; j < 4; j++) wma1 += prices[j] * (double)(4 - j);
        for(int j = 0; j < 9; j++) wma2 += prices[j] * (double)(9 - j);
        wma1 /= (4.0 * 5.0 / 2.0);
        wma2 /= (9.0 * 10.0 / 2.0);
        cachedHMA[idx] = 2.0 * wma1 - wma2;
     }
   }
   // Ichimoku Tenkan (M1, period=9)
   { double high[], low[]; ArraySetAsSeries(high, true); ArraySetAsSeries(low, true);
     if(CopyHigh(sym, PERIOD_M1, 0, 9, high) >= 9 && CopyLow(sym, PERIOD_M1, 0, 9, low) >= 9)
     {
        double hmax = high[0], lmin = low[0];
        for(int j = 1; j < 9; j++) { if(high[j] > hmax) hmax = high[j]; if(low[j] < lmin) lmin = low[j]; }
        cachedIchimokuTenkan[idx] = (hmax + lmin) / 2.0;
     }
   }
   // SAR (M1)
   { double b[]; ArraySetAsSeries(b, true);
     int h = iSAR(sym, PERIOD_M1, 0.02, 0.2);
     if(CopyBuffer(h, 0, 0, 1, b) > 0) cachedSAR[idx] = b[0];
     IndicatorRelease(h);
   }
   // DPO (M1, period=21)
   { double prices[]; ArraySetAsSeries(prices, true);
     if(CopyClose(sym, PERIOD_M1, 0, 21, prices) >= 21)
     {
        double sma = 0.0;
        for(int j = 0; j < 21; j++) sma += prices[j];
        sma /= 21.0;
        cachedDPO[idx] = prices[10] - sma;
     }
   }
   // Spread
   { double ask = SymbolInfoDouble(sym, SYMBOL_ASK);
     double bid = SymbolInfoDouble(sym, SYMBOL_BID);
     double pt  = SymbolInfoDouble(sym, SYMBOL_POINT);
     cachedSpread[idx] = pt > 0 ? (ask - bid) / pt : 0.0;
   }
   // Sentiment from server
   cachedSentiment[idx] = FetchSentiment(sym);
}

void ComputeHFTHelpers(const int idx, const string sym)
{
   if(idx < 0 || idx >= totalPairs)
      return;
   int want = HFTLastBars;
   if(want < 50)  want = 50;
   if(want > 400) want = 400;
   MqlRates rates[];
   ArraySetAsSeries(rates, true);
   int n = CopyRates(sym, PERIOD_M15, 0, want, rates);
   if(n < 8)
      return;

   if(UseVolumeHelpers)
     {
      HFTVolumePack vp;
      if(HFT_VolumeFromRates(rates, n, vp))
        {
         if(ArraySize(g_volPack) > idx)
            g_volPack[idx] = vp;
         cachedOBV[idx] = vp.obv;
         cachedVolumeDelta[idx] = vp.volume_delta;
         cachedADLine[idx] = vp.ad_line;
         cachedVolOsc[idx] = vp.vol_osc;
        }
     }

   double closesChrono[];
   ArrayResize(closesChrono, n);
   for(int i = 0; i < n; i++)
      closesChrono[i] = rates[n - 1 - i].close;

   double closesSeries[];
   ArrayResize(closesSeries, n);
   ArraySetAsSeries(closesSeries, true);
   for(int i = 0; i < n; i++)
      closesSeries[i] = rates[i].close;

   if(UseRegimeHelpers)
     {
      MqlDateTime ts;
      TimeToStruct(TimeGMT(), ts);
      HFTRegimePack rp;
      if(HFT_RegimeFromCloses(closesSeries, n, cachedATR[idx], cachedADX[idx], ts.hour, rp))
        {
         if(ArraySize(g_regPack) > idx)
            g_regPack[idx] = rp;
        }
     }

   if(UseMathMqh)
     {
      double rv = HFT_StdReturns(closesChrono, n);
      if(rv > 0.0)
         cachedRealizedVol[idx] = rv;
     }

   if(UseMatrixMqh && ArraySize(cachedZScore) > idx)
     {
      double z = 0.0, mean = 0.0, sd = 0.0;
      if(HFT_LastBarZScore(closesChrono, n, z, mean, sd))
         cachedZScore[idx] = z;
     }

   if((UseAlglibMqh || UseLinalgMqh) && ArraySize(cachedPxVolCorr) > idx)
     {
      double vols[];
      ArrayResize(vols, n);
      for(int i = 0; i < n; i++)
         vols[i] = (double)rates[n - 1 - i].tick_volume;
      cachedPxVolCorr[idx] = HFT_CorrAlglib(closesChrono, vols, n);
     }

   if(UseArrayMqh)
     {
      CArrayDouble ad;
      int k = MathMin(8, n);
      for(int i = 0; i < k; i++)
         ad.Add(rates[i].close);
     }

   if(UseSymbolMqh)
     {
      CSymbolInfo s;
      if(s.Name(sym))
        {
         s.RefreshRates();
         double pt = s.Point();
         if(pt > 0.0)
            cachedSpread[idx] = (s.Ask() - s.Bid()) / pt;
        }
     }
}

void ComputeM5Assist(const int idx, const string sym)
{
   if(!Saturn_InRange(idx, totalPairs) || !Saturn_InRange(idx, ArraySize(g_m5Pack)))
      return;
   SaturnM5Pack p;
   if(Saturn_M5FromRates(sym, p))
      g_m5Pack[idx] = p;
}

bool PassesM5Filter(const string sym, const int idx, const ENUM_ORDER_TYPE orderType)
{
   if(!UseM5Confirm)
      return true;
   if(!Saturn_InRange(idx, ArraySize(g_m5Pack)))
      return true;
   if(!g_m5Pack[idx].ok)
      ComputeM5Assist(idx, sym);
   if(!Saturn_PassesM5(g_m5Pack[idx], orderType))
     {
      Print("M5 filter: ", sym, " ", EnumToString(orderType),
            " rejected (impulse=", DoubleToString(g_m5Pack[idx].impulse, 6),
            " rsi=", DoubleToString(g_m5Pack[idx].rsi, 1),
            " confirm=", g_m5Pack[idx].confirm, ")");
      return false;
     }
   return true;
}

void RefreshH1Predict(const int idx, const string sym)
{
   if(!UseH1Predict && !UseSingleChartMTF)
      return;
   if(!Saturn_InRange(idx, ArraySize(g_h1Pack)))
      return;
   SaturnH1Pack p;
   Saturn_ReadH1Bridge(sym, StripBrokerSuffix(sym), H1PredictStaleSec, p);
   if((p.source == "" || !p.ok) && UseSingleChartMTF)
     {
      SaturnH1Pack loc;
      if(Saturn_H1FromRates(sym, loc))
         p = loc;
     }
   g_h1Pack[idx] = p;
}

bool PassesH1PredictFilter(const string sym, const int idx, const ENUM_ORDER_TYPE orderType)
{
   if(!UseH1Predict)
      return true;
   if(!Saturn_InRange(idx, ArraySize(g_h1Pack)))
      return !RequireH1Predict;
   if(!g_h1Pack[idx].ok)
      RefreshH1Predict(idx, sym);
   if(!Saturn_PassesH1Predict(g_h1Pack[idx], orderType, RequireH1Predict, AllowH1HoldPass))
     {
      Print("H1 Predict filter: ", sym, " ", EnumToString(orderType),
            " rejected (h1=", g_h1Pack[idx].signal,
            " conf=", DoubleToString(g_h1Pack[idx].conf, 4),
            " age=", g_h1Pack[idx].age,
            " src=", g_h1Pack[idx].source, ")");
      return false;
     }
   return true;
}

}

//+------------------------------------------------------------------+
//| Robust JSON Feature Builder                                      |
//+------------------------------------------------------------------+
string BuildPredictFeaturesJSON(const int idx, const ENUM_JSON_SCHEMA schema = SCHEMA_V1_APRIL2025)
{
   if(idx < 0 || idx >= totalPairs)
      return "{\"error\":\"bad_idx\"}";

   string sym = dynamicPairList[idx];
   double bid = SymbolInfoDouble(sym, SYMBOL_BID);
   double ask = SymbolInfoDouble(sym, SYMBOL_ASK);
   double mid = (bid + ask) * 0.5;

   // --- collect values with safety ---
   double price        = mid;
   double atr          = (ArraySize(cachedATR) > idx) ? cachedATR[idx] : 0.0;
   double ema_diff     = (ArraySize(cachedEMAFast) > idx && ArraySize(cachedEMASlow) > idx)
                         ? (cachedEMAFast[idx] - cachedEMASlow[idx]) : 0.0;
   double rsi          = (ArraySize(cachedRSI) > idx) ? cachedRSI[idx] : 50.0;
   double garch_vol    = (ArraySize(garchVolatility) > idx) ? garchVolatility[idx] : 0.0;
   double macd_diff    = (ArraySize(cachedMACD) > idx && ArraySize(cachedMACDSignal) > idx)
                         ? (cachedMACD[idx] - cachedMACDSignal[idx]) : 0.0;
   double vwap         = (ArraySize(cachedVWAP) > idx) ? cachedVWAP[idx] : mid;
   double price_vwap   = price - vwap;
   double bb_pos       = 0.5;
   if(ArraySize(cachedBBUpper) > idx && ArraySize(cachedBBLower) > idx &&
      cachedBBUpper[idx] > cachedBBLower[idx] + 1e-12)
      bb_pos = (price - cachedBBLower[idx]) / (cachedBBUpper[idx] - cachedBBLower[idx]);

//+------------------------------------------------------------------+
//| Automated schema migration (MQL5 side)                           |
//+------------------------------------------------------------------+
string MigrateAndBuildJSON(const int idx, const ENUM_JSON_SCHEMA target)
{
   // Always build the richest set we have (og_27_28), then let
   // the Python side migrate down if needed. This is the safest
   // direction (information-preserving).
   string full = BuildPredictFeaturesJSON(idx, SCHEMA_OG_27_28);

   // If the caller explicitly wants a smaller schema we can
   // still emit it directly (no round-trip needed).
   if(target == SCHEMA_OG_27_28)
      return full;

   // For the other two schemas we rebuild with the exact builder
   // (already implemented) so the JSON is minimal and correct.
   return BuildPredictFeaturesJSON(idx, target);
}

//+------------------------------------------------------------------+
//| CallAIAPI — full-feature call to golden model server              |
//| Sends all 28 config features + garch_vol so golden server can    |
//| route to xgb_6 (6-feat), ensemble_9 (9-feat), rf_28 (28-feat)   |
//+------------------------------------------------------------------+
string CallAIAPI(string symbol, int idx) {
   int refresh = (AiRefreshSec > 5) ? AiRefreshSec : 60;
   if (TimeCurrent() - last_api_call[idx] < refresh) {
      return last_good_signal[idx];
   }
   if (idx < 0 || idx >= totalPairs) {
      Print("Invalid index ", idx, " for ", symbol);
      return "hold";
   }

   ComputeM5Assist(idx, symbol);   // M5 is required — never omitted from payload

   string json = MigrateAndBuildJSON(idx, SCHEMA_OG_27_28);
   if(!ValidateFeatureSchema(json, SCHEMA_OG_27_28))
     {
      Print("Local feature schema invalid – aborting request");
      return last_good_signal[idx];
     }

   // splice saturn extras (M5 pack, H1, volume, regime, bar times) before the closing brace
   int jn = StringLen(json);
   if(jn > 0 && StringGetCharacter(json, jn - 1) == '}')
      json = StringSubstr(json, 0, jn - 1);
   if(UseVolumeHelpers && ArraySize(g_volPack) > idx)
     {
      json += ",\"volume\":"            + JsonNum(g_volPack[idx].volume, 2);
      json += ",\"volume_ratio\":"      + JsonNum(g_volPack[idx].volume_ratio, 6);
      json += ",\"up_volume_ratio\":"   + JsonNum(g_volPack[idx].up_volume_ratio, 6);
      json += ",\"volume_surge\":"      + JsonNum(g_volPack[idx].volume_surge, 1);
      json += ",\"volume_trend\":"      + JsonNum(g_volPack[idx].volume_trend, 1);
     }
   if(UseRegimeHelpers && ArraySize(g_regPack) > idx)
     {
      json += ",\"trend_filter\":"      + JsonNum(g_regPack[idx].trend_filter, 1);
      json += ",\"vol_regime\":"        + JsonNum(g_regPack[idx].vol_regime, 1);
      json += ",\"vol_percentile\":"    + JsonNum(g_regPack[idx].vol_percentile, 2);
      json += ",\"price_above_sma50\":" + JsonNum(g_regPack[idx].price_above_sma50, 1);
      json += ",\"price_above_sma200\":"+ JsonNum(g_regPack[idx].price_above_sma200, 1);
      json += ",\"sma_spread\":"        + JsonNum(g_regPack[idx].sma_spread, 8);
      json += ",\"session_asia\":"      + JsonNum(g_regPack[idx].session_asia, 1);
      json += ",\"session_london\":"    + JsonNum(g_regPack[idx].session_london, 1);
      json += ",\"session_ny\":"        + JsonNum(g_regPack[idx].session_ny, 1);
      json += ",\"regime_id\":"         + IntegerToString(g_regPack[idx].regime_id);
      json += ",\"regime_name\":"       + JsonStr(g_regPack[idx].regime_name);
     }
   if(UseMatrixMqh && ArraySize(cachedZScore) > idx)
      json += ",\"zscore_20\":" + JsonNum(cachedZScore[idx], 6);
   if((UseAlglibMqh || UseLinalgMqh) && ArraySize(cachedPxVolCorr) > idx)
      json += ",\"px_vol_corr\":" + JsonNum(cachedPxVolCorr[idx], 6);
   {
      double m5rsi = 50.0, m5ema = 0.0, m5mom = 0.0, m5vr = 1.0, m5imp = 0.0;
      int m5c = 0;
      if(Saturn_InRange(idx, ArraySize(g_m5Pack)))
        {
         m5rsi = g_m5Pack[idx].rsi;
         m5ema = g_m5Pack[idx].ema_diff;
         m5mom = g_m5Pack[idx].mom;
         m5vr  = g_m5Pack[idx].vol_ratio;
         m5imp = g_m5Pack[idx].impulse;
         m5c   = g_m5Pack[idx].confirm;
        }
      json += Saturn_JsonM5(m5rsi, m5ema, m5mom, m5vr, m5imp, m5c);
   }
   if(UseH1Predict && Saturn_InRange(idx, ArraySize(g_h1Pack)))
      json += Saturn_JsonH1(g_h1Pack[idx].signal, g_h1Pack[idx].conf,
                            g_h1Pack[idx].age, g_h1Pack[idx].source);
   json += ",\"chart_symbol\":" + JsonStr(StripBrokerSuffix(Symbol()));
   json += ",\"chart_tf\":"     + JsonStr(EnumToString(Period()));
   json += ",\"bar_time\":" + IntegerToString((int)iTime(symbol, Period(), 0));
   json += ",\"m15_bar\":" + IntegerToString((int)iTime(symbol, PERIOD_M15, 0));
   if(UseCorrNet)
      CorrNet_AppendJson(json);
   json += "}";

   Print("JSON sent to API for ", symbol, " schema=", SchemaToString(PredictFeatureSchema),
         " chars=", StringLen(json));

   string headers = "Content-Type: application/json\r\n";
   char post[];
   int jsonLength = StringLen(json);
   ArrayResize(post, jsonLength);
   StringToCharArray(json, post, 0, jsonLength, CP_UTF8);

   char result[];
   string response_headers;
   int maxRetries = 3;
   for (int retry = 0; retry < maxRetries; retry++) {
      int res = WebRequest("POST", AI_API_URL, headers, 10000, post, result, response_headers);
      string response = CharArrayToString(result, 0, -1, CP_UTF8);
      if (res == 200) {
         last_api_call[idx] = TimeCurrent();
         Print("API response for ", symbol, ": ", response);

         if(!JsonSchemaIsValidPredictResponse(response))
           {
            Print("Invalid predict schema from server");
            return "hold";
           }
         PredictResponse resp;
         if(!ParsePredictResponse(response, resp, MinAIConfidence, MaxAIConfidence) || !resp.valid)
           {
            Print("AI response failed schema validation");
            return "hold";
           }
         string rawSignal = resp.signal;
         double confidence = resp.confidence;
         if(rawSignal == "")
            rawSignal = "hold";

         last_good_signal[idx]    = rawSignal;
         last_good_signal_time[idx] = TimeCurrent();
         if (ArraySize(last_ai_confidence) > idx)
            last_ai_confidence[idx] = (rawSignal != "hold") ? confidence : 0.0;
         if (ArraySize(last_ai_vote_mode) > idx)
            last_ai_vote_mode[idx] = (resp.vote_mode != "") ? resp.vote_mode : resp.model;
         Print(symbol, " AI → ", rawSignal,
               " conf=", DoubleToString(confidence, 4),
               " model=", resp.model,
               " vote=", last_ai_vote_mode[idx]);
         return last_good_signal[idx];
      } else {
         int werr = GetLastError();
         Print("API call failed for ", symbol, ". HTTP: ", res, ", Error: ", werr,
               " ", Saturn_WebExplain(res, werr), ". Retry ", retry+1, "/", maxRetries);
         Sleep(1000);
      }
   }
   Print("API call failed for ", symbol, " after ", maxRetries, " attempts. Using last signal: ", last_good_signal[idx]);
   return last_good_signal[idx];
}

// Consecutive loss circuit breaker: pause trading after N losses in a row
bool IsConsecLossPaused() {
   if (!UseConsecLossBreaker) return false;
   if (g_consecutiveLosses >= ConsecLossPause) {
      if (g_consecLossPauseUntil == 0) {
         g_consecLossPauseUntil = TimeCurrent() + ConsecLossPauseHours * 3600;
         Print("CIRCUIT BREAKER: ", g_consecutiveLosses, " consecutive losses. Pausing until ",
               TimeToString(g_consecLossPauseUntil, TIME_DATE | TIME_MINUTES));
      }
      if (TimeCurrent() < g_consecLossPauseUntil) return true;
      // Pause expired, reset
      g_consecLossPauseUntil = 0;
      g_consecutiveLosses = 0;
      Print("Circuit breaker lifted. Resuming trading.");
   }
   return false;
}

// Session-based lot sizing multiplier
double GetSessionLotMultiplier() {
   if (!UseSessionSizing) return 1.0;
   MqlDateTime dt;
   TimeToStruct(TimeCurrent(), dt);
   int hour = dt.hour;
   int dow = dt.day_of_week;

   // Weekend (Saturday/Sunday) - crypto only
   if (dow == 0 || dow == 6) return SessionMultWeekend;

   // Asian session: 00:00-07:59 UTC
   if (hour >= 0 && hour < 8) return SessionMultAsian;

   // London session: 08:00-12:59 UTC
   if (hour >= 8 && hour < 13) return SessionMultLondon;

   // NY overlap + US session: 13:00-16:59 UTC
   if (hour >= 13 && hour < 17) return SessionMultNY;

   // Late US / pre-Asian: 17:00-23:59 UTC
   return SessionMultAsian;
}

// Round number avoidance: skip trades near psychological levels
bool IsNearRoundNumber(string sym, double price) {
   if (!UseRoundNumberFilter) return false;
   if (price <= 0) return false;

   double roundLevel = 0;
   // Determine round number interval based on price magnitude
   if (price > 10000)       roundLevel = 5000;    // BTC: 80000, 85000, 90000
   else if (price > 1000)   roundLevel = 500;     // ETH: 1500, 2000, 2500
   else if (price > 100)    roundLevel = 50;      // XAU: 2000, 2050
   else if (price > 10)     roundLevel = 5;       // JPY: 150, 155
   else                     roundLevel = 0.05;    // EUR: 1.05, 1.10

   double nearest = MathRound(price / roundLevel) * roundLevel;
   double distPct = MathAbs(price - nearest) / price;

   if (distPct < RoundNumberPct) {
      Print("Round number filter: ", sym, " price ", DoubleToString(price, (int)SymbolInfoInteger(sym, SYMBOL_DIGITS)),
            " too close to ", DoubleToString(nearest, 2), " (", DoubleToString(distPct * 100, 2), "%)");
      return true;
   }
   return false;
}

// Multi-timeframe confirmation: check H1 EMA direction
// Returns 1 for bullish H1, -1 for bearish H1, 0 for neutral
int GetH1TrendDirection(string sym, int idx) {
   if (!UseMultiTFConfirm) return 0;  // disabled = no filter
   if (idx < 0 || idx >= totalPairs) return 0;
   if (h1EmaHandles[idx] == INVALID_HANDLE) return 0;

   double ema[2];
   if (CopyBuffer(h1EmaHandles[idx], 0, 0, 2, ema) < 2) return 0;

   // EMA rising = bullish, falling = bearish
   if (ema[0] > ema[1]) return 1;   // current > previous = uptrend
   if (ema[0] < ema[1]) return -1;  // current < previous = downtrend
   return 0;
}

// Check if trade direction aligns with H1 trend
bool PassesMultiTFFilter(string sym, int idx, ENUM_ORDER_TYPE orderType) {
   if (!UseMultiTFConfirm) return true;
   int h1Trend = GetH1TrendDirection(sym, idx);
   if (h1Trend == 0) return true;  // no data = pass

   // Buy must align with bullish H1, sell with bearish H1
   if (orderType == ORDER_TYPE_BUY && h1Trend < 0) {
      Print("Multi-TF filter: ", sym, " BUY rejected - H1 trend is bearish");
      return false;
   }
   if (orderType == ORDER_TYPE_SELL && h1Trend > 0) {
      Print("Multi-TF filter: ", sym, " SELL rejected - H1 trend is bullish");
      return false;
   }
   return true;
}

double CalculatePairCorrelation(string pair1, string pair2, int period) {
   double prices1[], prices2[];
   ArraySetAsSeries(prices1, true); ArraySetAsSeries(prices2, true);
   if (CopyClose(pair1, PERIOD_M15, 0, period, prices1) < period ||
       CopyClose(pair2, PERIOD_M15, 0, period, prices2) < period) return 0.0;
   double mean1 = 0.0, mean2 = 0.0;
   for(int i = 0; i < period; i++) { mean1 += prices1[i]; mean2 += prices2[i]; }
   mean1 /= period; mean2 /= period;
   double cov = 0.0, var1 = 0.0, var2 = 0.0;
   for(int i = 0; i < period; i++) {
      double d1 = prices1[i] - mean1, d2 = prices2[i] - mean2;
      cov += d1 * d2; var1 += d1 * d1; var2 += d2 * d2;
   }
   if(var1 == 0.0 || var2 == 0.0) return 0.0;
   return cov / (MathSqrt(var1) * MathSqrt(var2));
}

bool CheckCorrelationLimit(FXJEFE_CandidateTrade &candidate) {
   for (int i = 0; i < PositionsTotal(); i++) {
      if (positionInfo.SelectByTicket(PositionGetTicket(i))) {
         string existingPair = positionInfo.Symbol();
         if (existingPair != candidate.pair) {
            double corr = CalculatePairCorrelation(candidate.pair, existingPair, 20);
            if (MathAbs(corr) > MaxCorrelation) {
               Print("Trade rejected: High correlation (", corr, ") between ", candidate.pair, " and ", existingPair);
               return false;
            }
         }
      }
   }
   return true;
}

bool QueueAiCandidate(const string sym, const int i, const string aiSignal,
                      FXJEFE_CandidateTrade &candidates[])
{
   if(aiSignal != "buy" && aiSignal != "sell")
      return false;
   double conf = (ArraySize(last_ai_confidence) > i) ? last_ai_confidence[i] : 0.0;
   if(HolyOnlyAboveGate && !Saturn_ConfInBand(conf, MinAIConfidence, MaxAIConfidence))
     {
      Print("AI print ", sym, " ", aiSignal, " conf=", DoubleToString(conf, 4),
            " not in holy band — no trade");
      return false;
     }
   ENUM_SYMBOL_TRADE_MODE tm = (ENUM_SYMBOL_TRADE_MODE)SymbolInfoInteger(sym, SYMBOL_TRADE_MODE);
   if(tm != SYMBOL_TRADE_MODE_FULL && tm != SYMBOL_TRADE_MODE_LONGONLY &&
      tm != SYMBOL_TRADE_MODE_SHORTONLY)
     {
      Print("AI print ", sym, " ", aiSignal, " — market closed, skip");
      return false;
     }
   if(tm == SYMBOL_TRADE_MODE_LONGONLY && aiSignal == "sell")
     {
      Print("AI print ", sym, " sell — LONGONLY, skip");
      return false;
     }
   if(tm == SYMBOL_TRADE_MODE_SHORTONLY && aiSignal == "buy")
     {
      Print("AI print ", sym, " buy — SHORTONLY, skip");
      return false;
     }

   FXJEFE_CandidateTrade c;
   c.pair = sym;
   c.strategyName = "AI_Signal";
   c.orderType  = (aiSignal == "buy") ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   c.openPrice  = (c.orderType == ORDER_TYPE_BUY) ? SymbolInfoDouble(sym, SYMBOL_ASK)
                                                 : SymbolInfoDouble(sym, SYMBOL_BID);
   double atr = (ArraySize(cachedATR) > i && cachedATR[i] > 0.0) ? cachedATR[i]
                : SymbolInfoDouble(sym, SYMBOL_POINT) * 100.0;
   c.stopLoss   = (c.orderType == ORDER_TYPE_BUY) ? c.openPrice - atr * 2.0
                                                 : c.openPrice + atr * 2.0;
   c.takeProfit = (c.orderType == ORDER_TYPE_BUY) ? c.openPrice + atr * 3.0
                                                 : c.openPrice - atr * 3.0;
   string vmode = (ArraySize(last_ai_vote_mode) > i) ? last_ai_vote_mode[i] : "";
   c.lotSize    = CalculateLotSize(sym, c.stopLoss, c.openPrice, c.orderType, conf, vmode);
   double slDist = MathAbs(c.openPrice - c.stopLoss);
   if(slDist > 0.0)
      c.takeProfit = (c.orderType == ORDER_TYPE_BUY) ? c.openPrice + slDist * 1.5
                                                    : c.openPrice - slDist * 1.5;
   if(c.lotSize <= 0.0)
     {
      double vmin = SymbolInfoDouble(sym, SYMBOL_VOLUME_MIN);
      if(vmin <= 0.0) vmin = 0.01;
      c.lotSize = vmin;
      Print("AI print ", sym, " lot was 0 — using broker min ", DoubleToString(vmin, 3));
     }
   if(UseCorrNet && CorrNet_IndexOf(sym) >= 0)
      conf = MathMin(0.9888, conf + 0.04 * CorrNet_ScoreOf(sym) *
             ((c.orderType == ORDER_TYPE_BUY) ? 1.0 : -1.0));
   c.momentumScore = (conf > 0.0) ? conf : MinAIConfidence;
   ArrayAppend(candidates, c);
   Print("AI TRADE queued ", sym, " ", aiSignal,
         " conf=", DoubleToString(c.momentumScore, 4),
         " lot=", DoubleToString(c.lotSize, 3),
         " vote=", vmode,
         " (H1/M5/round/session do not veto AI prints)");
   return true;
}

void ScanAllStrategies(FXJEFE_CandidateTrade &candidates[]) {
   ArrayResize(candidates, 0);

   // Circuit breaker: skip all scanning if paused after consecutive losses
   if (IsConsecLossPaused()) return;

   for (int i = 0; i < totalPairs; i++) {
      string sym = dynamicPairList[i];

      // Skip if broker has closed this symbol for trading right now
      ENUM_SYMBOL_TRADE_MODE tradeMode = (ENUM_SYMBOL_TRADE_MODE)SymbolInfoInteger(sym, SYMBOL_TRADE_MODE);
      if (tradeMode != SYMBOL_TRADE_MODE_FULL && tradeMode != SYMBOL_TRADE_MODE_LONGONLY &&
          tradeMode != SYMBOL_TRADE_MODE_SHORTONLY) continue;

      FXJEFE_CandidateTrade c;
      // Band-valid AI prints become candidates first — mentor-aligned, no H1/M5 veto.
      if (SignalMode == AI_Only || SignalMode == Both) {
         string aiSignal = UseAISignals ? CallAIAPI(sym, i) : "hold";
         if (aiSignal == "buy" || aiSignal == "sell")
            QueueAiCandidate(sym, i, aiSignal, candidates);
      }

      // Skip if outside allowed trading hours for this symbol (night / session transitions)
      if (!IsTradingAllowedForSymbol(sym)) continue;

      if (!CheckLiquidity(sym)) continue;

      // Round number filter: skip if price is too close to psychological level
      double currentBid = SymbolInfoDouble(sym, SYMBOL_BID);
      if (IsNearRoundNumber(sym, currentBid)) continue;
      if (SignalMode == Strategies_Only || SignalMode == Both) {
         if (HasMicroBreakoutSignal(sym, i, c) && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasPullbackTrendSignal(sym, i, c) && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasICTKillZoneSignal(sym, i, c)   && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasPO3Signal(sym, i, c)            && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasPsychLevelsSignal(sym, i, c)   && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasStatArbitrageSignal(sym, i, c) && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
         if (HasCarryTradeSignal(sym, i, c)    && c.lotSize > 0 && CheckCorrelationLimit(c) && PassesMultiTFFilter(sym, i, c.orderType) && PassesM5Filter(sym, i, c.orderType) && PassesH1PredictFilter(sym, i, c.orderType)) ArrayAppend(candidates, c);
      }
   }
}

void PickAndOpenBestTrades(FXJEFE_CandidateTrade &candidates[]) {
   int size = ArraySize(candidates);
   for (int i = 0; i < size - 1; i++) {
      for (int j = 0; j < size - i - 1; j++) {
         if (candidates[j].momentumScore < candidates[j+1].momentumScore) {
            FXJEFE_CandidateTrade temp = candidates[j]; candidates[j] = candidates[j+1]; candidates[j+1] = temp;
         }
      }
   }
   int tradesToOpen = MathMin(ArraySize(candidates), MaxOpenTrades - PositionsTotal());
   double sessionMult = GetSessionLotMultiplier();

   for (int i = 0; i < tradesToOpen; i++) {
      string cpair = candidates[i].pair;

      // Re-check market is open before sending (may have closed since scan)
      ENUM_SYMBOL_TRADE_MODE tm = (ENUM_SYMBOL_TRADE_MODE)SymbolInfoInteger(cpair, SYMBOL_TRADE_MODE);
      if (tm != SYMBOL_TRADE_MODE_FULL && tm != SYMBOL_TRADE_MODE_LONGONLY &&
          tm != SYMBOL_TRADE_MODE_SHORTONLY) {
         Print("SKIP: ", cpair, " market closed at execution time");
         continue;
      }

      // Session sizing — snap to broker min/step only (no 0.01/0.1 hard lock)
      double stepLot   = SymbolInfoDouble(cpair, SYMBOL_VOLUME_STEP);
      double brokerMin = SymbolInfoDouble(cpair, SYMBOL_VOLUME_MIN);
      if (stepLot <= 0.0) stepLot = brokerMin;
      if (stepLot <= 0.0) stepLot = 0.01;
      if (stepLot <= 0.0) { Print("SKIP lot snap: ", cpair, " zero volume step"); continue; }
      double hardMin = brokerMin;
      if (hardMin <= 0.0) hardMin = stepLot;
      int volDigits = 2;
      if (stepLot < 0.01 - 1e-12) volDigits = 3;
      else if (stepLot >= 1.0 - 1e-12) volDigits = 0;
      hardMin = NormalizeDouble(hardMin, volDigits);
      candidates[i].lotSize *= sessionMult;
      candidates[i].lotSize = NormalizeDouble(
         MathMax(hardMin, MathFloor(candidates[i].lotSize / stepLot + 1e-12) * stepLot), volDigits);

      if (candidates[i].lotSize <= 0) continue;  // micro protection filtered this out
      VerifySymbolSpecs(cpair, candidates[i].lotSize);
      double leverage = CalculateTrueLeverage(candidates[i].lotSize, candidates[i].pair);
      // Guard: $0 equity → leverage 0 → never divide by leverage
      if (leverage > 0.0 && g_leverageCap > 0.0 && leverage > g_leverageCap)
         candidates[i].lotSize *= g_leverageCap / leverage;
      double slippage = GetDynamicSlippage(candidates[i].pair, ArraySearchString(dynamicPairList, candidates[i].pair));
      double pt = SymbolInfoDouble(candidates[i].pair, SYMBOL_POINT);
      if (pt <= 0.0) pt = 0.00001;
      trade.SetDeviationInPoints((long)(slippage / pt));
      if (trade.PositionOpen(candidates[i].pair, candidates[i].orderType, candidates[i].lotSize,
                             candidates[i].openPrice, candidates[i].stopLoss, candidates[i].takeProfit,
                             candidates[i].strategyName)) {
         g_dailyTradesCount++;
         g_tradingDayActive = true;
         if (sessionMult < 1.0)
            Print("Session sizing: lot reduced by ", DoubleToString((1.0 - sessionMult) * 100, 0), "% (mult=", DoubleToString(sessionMult, 2), ")");
         Print("AI TRADE opened ", candidates[i].pair, " ", candidates[i].strategyName,
               " lot=", DoubleToString(candidates[i].lotSize, 3),
               " conf=", DoubleToString(candidates[i].momentumScore, 4));
         LogTradeOpen(candidates[i]);
      }
      else
        {
         Print("AI TRADE FAIL ", candidates[i].pair, " ", candidates[i].strategyName,
               " ret=", trade.ResultRetcode(), " ", trade.ResultRetcodeDescription(),
               " err=", GetLastError());
        }
   }
}

double CalculateVaR() {
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   if (equity <= 0.0) equity = g_initialBalance;
   if (equity <= 0.0) return 0.0;  // unfunded / zero-equity account — no VaR% (avoids zero divide)
   double totalRisk = 0.0;
   for (int i = 0; i < PositionsTotal(); i++) {
      if (positionInfo.SelectByTicket(PositionGetTicket(i))) {
         string sym = positionInfo.Symbol();
         int idx = ArraySearchString(dynamicPairList, sym);
         if (idx >= 0) totalRisk += positionInfo.Volume() * cachedATR[idx] * SymbolInfoDouble(sym, SYMBOL_TRADE_TICK_VALUE);
      }
   }
   return (totalRisk / equity) * 100.0;
}

void CheckRiskLimits() {
   // Full risk mode (MaxTradeRiskPct>=100): no VaR hard halt
   if (MaxTradeRiskPct >= 100.0) return;
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   if (equity <= 0.0 && g_initialBalance <= 0.0) return;
   double var = CalculateVaR();
   if (var > MaxVaR_Pct) { Print("VaR limit exceeded: ", var, "% > ", MaxVaR_Pct, "%"); tradingEnabled = false; }
}

      last_good_signal_time[i]=0; last_ai_confidence[i]=0.0; last_ai_vote_mode[i]=""; tokyoHigh[i]=0.0; tokyoLow[i]=0.0;
      lastKillZoneStart[i]=0; firstHigh[i]=0.0; firstLow[i]=0.0;
   }

   // Init M15 indicator handles
   for (int i = 0; i < totalPairs; i++) {
      string sym = dynamicPairList[i];
      atrHandles[i]         = iATR(sym, PERIOD_M15, ATR_Period);
      emaFastHandles[i]     = iMA(sym, PERIOD_M15, EMA_Fast_Period, 0, MODE_EMA, PRICE_CLOSE);
      emaSlowHandles[i]     = iMA(sym, PERIOD_M15, EMA_Slow_Period, 0, MODE_EMA, PRICE_CLOSE);
      rsiHandles[i]         = iRSI(sym, PERIOD_M15, RSI_Period, PRICE_CLOSE);
      bbHandles[i]          = iBands(sym, PERIOD_M15, BB_Period, 0, BB_Deviation, PRICE_CLOSE);
      stochasticHandles[i]  = iStochastic(sym, PERIOD_M15, Stochastic_K, Stochastic_D, Stochastic_Slow, MODE_SMA, STO_LOWHIGH);
      macdHandles[i]        = iMACD(sym, PERIOD_M15, MACD_Fast, MACD_Slow, MACD_Signal, PRICE_CLOSE);
      adxHandles[i]         = iADX(sym, PERIOD_M15, 14);
      // H1 EMA for multi-timeframe confirmation
      h1EmaHandles[i] = iMA(sym, PERIOD_H1, 50, 0, MODE_EMA, PRICE_CLOSE);

      if (atrHandles[i]==INVALID_HANDLE || emaFastHandles[i]==INVALID_HANDLE || emaSlowHandles[i]==INVALID_HANDLE ||
          rsiHandles[i]==INVALID_HANDLE || bbHandles[i]==INVALID_HANDLE || stochasticHandles[i]==INVALID_HANDLE ||
          macdHandles[i]==INVALID_HANDLE || adxHandles[i]==INVALID_HANDLE) {
         Print("WARNING: Failed M15 indicators for ", sym, " (Error: ", GetLastError(), ") — removing from pair list");
         ArrayRemove(dynamicPairList, i);
         totalPairs = ArraySize(dynamicPairList);
         // Re-init needed: resize arrays and restart loop
         ArrayResize(atrHandles, totalPairs);      ArrayResize(emaFastHandles, totalPairs);  ArrayResize(emaSlowHandles, totalPairs);
         ArrayResize(rsiHandles, totalPairs);      ArrayResize(bbHandles, totalPairs);       ArrayResize(stochasticHandles, totalPairs);
         ArrayResize(macdHandles, totalPairs);     ArrayResize(adxHandles, totalPairs);      ArrayResize(h1EmaHandles, totalPairs);
         i--;  // retry this index (now holds next symbol)
         continue;
      }
      if (h1EmaHandles[i]==INVALID_HANDLE) {
         Print("Warning: H1 EMA handle failed for ", sym, " - multi-TF filter disabled for this pair");
      }
   }

   if (totalPairs == 0) {
      Print("ERROR: No valid pairs with indicator data. EA cannot run.");
      return INIT_FAILED;
   }
   Print("Initialized ", totalPairs, " pairs successfully.");

   // Determine initial balance: Custom > Enum > Broker (never leave at $0 if equity exists)
   double brokerBal = AccountInfoDouble(ACCOUNT_BALANCE);
   double brokerEq  = AccountInfoDouble(ACCOUNT_EQUITY);
   if (AccountSize == Acct_Custom) {
      if (CustomAccountSize > 0)
         g_initialBalance = CustomAccountSize;
      else
         g_initialBalance = brokerBal;
      // Fallback: balance can report 0 on some demos / before history sync — use equity
      if (g_initialBalance <= 0.0)
         g_initialBalance = MathMax(brokerEq, brokerBal);
      Print("Custom account size: $", DoubleToString(g_initialBalance, 2),
            " | Broker bal=$", DoubleToString(brokerBal, 2),
            " eq=$", DoubleToString(brokerEq, 2),
            " | leverage 1:", (int)AccountInfoInteger(ACCOUNT_LEVERAGE));
   } else {
      g_initialBalance = (double)AccountSize;
      if (MathAbs(brokerBal - g_initialBalance) > 1.0)
         Print("Warning: Account balance $", DoubleToString(brokerBal, 2),
               " differs from AccountSize $", DoubleToString(g_initialBalance, 2));
   }
   if (g_initialBalance <= 0.0)
      Print("WARNING: Account equity/balance is $0 — DD stop disabled. Fund the account or set CustomAccountSize.");

   if (PhaseType == Phase_Live) {
      g_totalProfitTarget = 0.0;
      Print("LIVE MODE: No profit target. DailyDD=", (UseDailyDDStop ? "ON" : "OFF"),
            " TotalDD=", (UseTotalDDStop ? "ON" : "OFF"),
            " MaxTradeRiskPct=", DoubleToString(MaxTradeRiskPct, 1), "%");
   } else {
      g_totalProfitTarget = g_initialBalance * (TotalProfitTargetPct / 100.0);
   }
   g_previousDayEquity  = MathMax(g_initialBalance, brokerEq);
   g_maxDailyLoss       = (UseDailyDDStop && g_previousDayEquity > 0.0)
                          ? (g_previousDayEquity * (DailyDDLimitPct / 100.0)) : 0.0;
   g_maxBalance         = MathMax(brokerBal, g_initialBalance);
   g_dailyStartEquity   = (brokerEq > 0.0) ? brokerEq : g_previousDayEquity;
   g_lastEquityCheck    = g_dailyStartEquity;
   {
      MqlDateTime _ts; TimeToStruct(TimeCurrent(), _ts);
      g_lastDayReset   = TimeCurrent() - (_ts.hour * 3600 + _ts.min * 60 + _ts.sec);
      g_lastEquityTime = TimeCurrent();
   }
   g_dailyProfit        = 0.0;
   g_tradingDaysCount   = 0;
   ArrayResize(g_tradingDays, 0);
   Print("Risk: MaxTradeRiskPct=", DoubleToString(MaxTradeRiskPct, 1),
         "% | DailyDD=", (UseDailyDDStop ? "ON" : "OFF"),
         " | TotalDD=", (UseTotalDDStop ? "ON" : "OFF"));
   Saturn_StartKeepaliveTimer();   // 1s wall-clock — bridge never sleeps

   // Force initial M15 cache fill
   for (int i = 0; i < totalPairs; i++) {
      string sym = dynamicPairList[i];
      double temp[1];
      if (CopyBuffer(atrHandles[i],         0, 0, 1, temp) > 0) cachedATR[i]        = temp[0];
      if (CopyBuffer(emaFastHandles[i],      0, 0, 1, temp) > 0) cachedEMAFast[i]   = temp[0];
      if (CopyBuffer(emaSlowHandles[i],      0, 0, 1, temp) > 0) cachedEMASlow[i]   = temp[0];
      if (CopyBuffer(rsiHandles[i],          0, 0, 1, temp) > 0) cachedRSI[i]       = temp[0];
      if (CopyBuffer(bbHandles[i],           1, 0, 1, temp) > 0) cachedBBUpper[i]   = temp[0];
      if (CopyBuffer(bbHandles[i],           2, 0, 1, temp) > 0) cachedBBLower[i]   = temp[0];
      if (CopyBuffer(stochasticHandles[i],   0, 0, 1, temp) > 0) cachedStochK[i]    = temp[0];
      if (CopyBuffer(stochasticHandles[i],   1, 0, 1, temp) > 0) cachedStochD[i]    = temp[0];
      if (CopyBuffer(macdHandles[i],         0, 0, 1, temp) > 0) cachedMACD[i]      = temp[0];
      if (CopyBuffer(macdHandles[i],         1, 0, 1, temp) > 0) cachedMACDSignal[i]= temp[0];
      if (CopyBuffer(adxHandles[i],          0, 0, 1, temp) > 0) cachedADX[i]       = temp[0];
      CalculateVWAP(i, sym);
      ComputeExtendedFeatures(i, sym);   // FIX: init M1 model features
      ComputeHFTHelpers(i, sym);
      ComputeM5Assist(i, sym);
      RefreshH1Predict(i, sym);
   }
   indicatorsInitialized = true;
   Print("All indicators initialized on startup. M5 assist=", (UseM5Confirm || UseM5LivePredict),
         " H1 Predict=", UseH1Predict);
   ExportChartLastBars();
   LogFeatures();
   g_lastHealthPing = TimeLocal();
   Saturn_HealthPing(AI_HEALTH_URL, 0);
   return INIT_SUCCEEDED;
}

void OnDeinit(const int reason) {
   for (int i = 0; i < totalPairs; i++) {
      IndicatorRelease(atrHandles[i]);  IndicatorRelease(emaFastHandles[i]); IndicatorRelease(emaSlowHandles[i]);
      IndicatorRelease(rsiHandles[i]);  IndicatorRelease(bbHandles[i]);      IndicatorRelease(stochasticHandles[i]);
      IndicatorRelease(macdHandles[i]); IndicatorRelease(adxHandles[i]);
   }
   EventKillTimer();
   if (g_apiPipeOpen) apiPipe.Close();
}

void OnTimer() {
   if (g_timerRunning) return;
   g_timerRunning = true;
   int pingEvery = (HealthPingSec > 30) ? HealthPingSec : 300;
   Saturn_HealthPing(AI_HEALTH_URL, pingEvery);
   Saturn_TouchAlive();
   // Bridge keepalive even when the feed is down — this box runs 24/7.
   for (int bi = 0; bi < totalPairs; bi++) {
      if (!Saturn_InRange(bi, ArraySize(dynamicPairList))) break;
      string bsym = dynamicPairList[bi];
      string bbase = StripBrokerSuffix(bsym);
      RefreshH1Predict(bi, bsym);
      string hsig = "hold";
      double hconf = 0.0;
      if (Saturn_InRange(bi, ArraySize(g_h1Pack)) && g_h1Pack[bi].source != "")
        {
         hsig = g_h1Pack[bi].signal;
         hconf = g_h1Pack[bi].conf;
        }
      Saturn_WriteTfBridge("H1", bbase, bsym, hsig, hconf);
   }
   if (!TerminalInfoInteger(TERMINAL_CONNECTED)) { g_timerRunning = false; return; }
   double currentEquity = AccountInfoDouble(ACCOUNT_EQUITY);
   MqlDateTime timeStruct; TimeToStruct(TimeCurrent(), timeStruct);
   datetime currentDay = TimeCurrent() - (timeStruct.hour * 3600 + timeStruct.min * 60 + timeStruct.sec);

   if (currentDay > g_lastDayReset) {
      g_previousDayEquity = (currentEquity > 0.0) ? currentEquity : g_previousDayEquity;
      g_maxDailyLoss      = (UseDailyDDStop && g_previousDayEquity > 0.0)
                           ? (g_previousDayEquity * (DailyDDLimitPct / 100.0)) : 0.0;
      g_dailyStartEquity  = g_previousDayEquity;
      g_dailyProfit       = 0.0;
      g_lastDayReset      = currentDay;
      if (g_tradingDayActive) {
         string dayStr = TimeToString(currentDay, TIME_DATE);
         if (ArraySearchString(g_tradingDays, dayStr) == -1) {
            int sz = ArraySize(g_tradingDays); ArrayResize(g_tradingDays, sz + 1);
            g_tradingDays[sz] = dayStr; g_tradingDaysCount++;
         }
      }
      g_tradingDayActive  = false;
      g_dailyTradesCount  = 0;
      tradingEnabled      = true;
   } else {
      g_dailyProfit = currentEquity - g_dailyStartEquity;
   }

   // Policy: never flatten / mass-close on daily or total DD (flags ignored).
   if (UseDailyDDStop && g_maxDailyLoss > 0.01 && g_dailyProfit <= -g_maxDailyLoss)
      Print("Daily DD note (NO flatten): ", DoubleToString(-g_dailyProfit, 2),
            " vs cap ", DoubleToString(g_maxDailyLoss, 2));
   if (UseTotalDDStop && g_initialBalance > 0.01)
     {
      double totalDrawdown = g_initialBalance - currentEquity;
      if (totalDrawdown >= g_initialBalance * 0.10)
         Print("Total DD note (NO flatten): $", DoubleToString(totalDrawdown, 2));
     }
   if (g_totalProfitTarget > 0 && currentEquity >= g_initialBalance + g_totalProfitTarget && !g_profitTargetReached)
      g_profitTargetReached = true;  // mark only — do not close positions

   bool newM5  = NewBar(PERIOD_M5);
   bool newM15 = NewBar(PERIOD_M15);
   bool newH1  = NewBar(PERIOD_H1);

   if (newH1 || UseH1Predict) {
      for (int i = 0; i < totalPairs; i++) {
         if (!Saturn_InRange(i, ArraySize(dynamicPairList))) break;
         RefreshH1Predict(i, dynamicPairList[i]);
      }
      if(newH1)
         Print("H1 Predict bridge refreshed for ", totalPairs, " pairs.");
   }

   if (newM5) {
      for (int i = 0; i < totalPairs; i++) {
         if (!Saturn_InRange(i, ArraySize(dynamicPairList))) break;
         ComputeM5Assist(i, dynamicPairList[i]);
      }
      Print("M5 live assist updated for ", totalPairs, " pairs.");
   }
   if(UseCorrNet && (newM5 || newM15 || newH1 || g_cnStamp == 0))
     {
      CorrNet_Update(CorrNetBars);
      Print("CorrNet M15 corr BTC-XRP=", DoubleToString(g_cnCorr[0], 3),
            " BTC-XAU=", DoubleToString(g_cnCorr[1], 3),
            " XRP-XAU=", DoubleToString(g_cnCorr[2], 3),
            " (XAU not in dynamicPairList)");
     }

   if (newM15) {
      TimeToStruct(TimeCurrent(), timeStruct);
      if (timeStruct.hour == 0 && timeStruct.min < 15) {
         for (int i = 0; i < totalPairs; i++) {
            tokyoHigh[i] = iHigh(dynamicPairList[i], PERIOD_M15, 0);
            tokyoLow[i]  = iLow(dynamicPairList[i],  PERIOD_M15, 0);
         }
      }
      if (timeStruct.hour == 7 && timeStruct.min < 15) {
         for (int i = 0; i < totalPairs; i++) {
            firstHigh[i] = iHigh(dynamicPairList[i], PERIOD_M15, 0);
            firstLow[i]  = iLow(dynamicPairList[i],  PERIOD_M15, 0);
         }
      }
      for (int i = 0; i < totalPairs; i++) {
         if (!Saturn_InRange(i, ArraySize(dynamicPairList))) break;
         if (!Saturn_InRange(i, ArraySize(atrHandles))) break;
         if (iBars(dynamicPairList[i], PERIOD_M15) < 50) continue;
         string sym = dynamicPairList[i];
         double temp[];
         if (Saturn_SafeCopyBuffer(atrHandles[i],         0, 0, 1, temp) > 0) cachedATR[i]         = temp[0]; else Print("Failed ATR ", sym);
         if (Saturn_SafeCopyBuffer(emaFastHandles[i],      0, 0, 1, temp) > 0) cachedEMAFast[i]    = temp[0]; else Print("Failed EMA Fast ", sym);
         if (Saturn_SafeCopyBuffer(emaSlowHandles[i],      0, 0, 1, temp) > 0) cachedEMASlow[i]    = temp[0]; else Print("Failed EMA Slow ", sym);
         if (Saturn_SafeCopyBuffer(rsiHandles[i],          0, 0, 1, temp) > 0) cachedRSI[i]        = temp[0]; else Print("Failed RSI ", sym);
         if (Saturn_SafeCopyBuffer(bbHandles[i],           1, 0, 1, temp) > 0) cachedBBUpper[i]    = temp[0]; else Print("Failed BB Upper ", sym);
         if (Saturn_SafeCopyBuffer(bbHandles[i],           2, 0, 1, temp) > 0) cachedBBLower[i]    = temp[0]; else Print("Failed BB Lower ", sym);
         if (Saturn_SafeCopyBuffer(stochasticHandles[i],   0, 0, 1, temp) > 0) cachedStochK[i]     = temp[0]; else Print("Failed Stoch K ", sym);
         if (Saturn_SafeCopyBuffer(stochasticHandles[i],   1, 0, 1, temp) > 0) cachedStochD[i]     = temp[0]; else Print("Failed Stoch D ", sym);
         if (Saturn_SafeCopyBuffer(macdHandles[i],         0, 0, 1, temp) > 0) cachedMACD[i]       = temp[0]; else Print("Failed MACD ", sym);
         if (Saturn_SafeCopyBuffer(macdHandles[i],         1, 0, 1, temp) > 0) cachedMACDSignal[i] = temp[0]; else Print("Failed MACD Signal ", sym);
         if (Saturn_SafeCopyBuffer(adxHandles[i],          0, 0, 1, temp) > 0) cachedADX[i]        = temp[0]; else Print("Failed ADX ", sym);
         CalculateVWAP(i, sym);
         ComputeExtendedFeatures(i, sym);   // FIX: update M1 model features each bar
         ComputeHFTHelpers(i, sym);
         ComputeM5Assist(i, sym);
         RefreshH1Predict(i, sym);
      }
      UpdateGARCHVolatility();
      indicatorsInitialized = true;
      Print("Indicators updated for all pairs.");
      ExportChartLastBars();
      LogFeatures();
      CheckRiskLimits();
      if (tradingEnabled && IsTradingAllowedNow()) {
         int total = PositionsTotal();
         if (total < MaxOpenTrades && (!UseMaxDailyTrades || g_dailyTradesCount < MaxDailyTrades)) {
            FXJEFE_CandidateTrade cands[];
            ScanAllStrategies(cands);
            Print("Scan queued ", ArraySize(cands), " candidate(s) — AI prints open without H1/M5 veto");
            if (ArraySize(cands) > 0) PickAndOpenBestTrades(cands);
         }
      }
   }
   else if (newM5 && UseM5LivePredict && tradingEnabled && IsTradingAllowedNow()) {
      // Mid-M15: live M5 refresh of the chart pair (and all pairs) to help current M15 trade
      string chart = Symbol();
      int cidx = ArraySearchString(dynamicPairList, chart);
      if (Saturn_InRange(cidx, totalPairs)) {
         last_api_call[cidx] = 0;   // allow fresh /predict with latest M5
         ComputeHFTHelpers(cidx, dynamicPairList[cidx]);
      }
      int total = PositionsTotal();
      if (total < MaxOpenTrades && (!UseMaxDailyTrades || g_dailyTradesCount < MaxDailyTrades)) {
         FXJEFE_CandidateTrade cands[];
         ScanAllStrategies(cands);
         Print("M5 scan queued ", ArraySize(cands), " candidate(s)");
         if (ArraySize(cands) > 0) PickAndOpenBestTrades(cands);
      }
   }
   MultiPartialExit();
   g_timerRunning = false;
}

void OnTradeTransaction(const MqlTradeTransaction& trans, const MqlTradeRequest& request, const MqlTradeResult& result) {
   if (trans.type == TRADE_TRANSACTION_DEAL_ADD && (trans.deal_type == DEAL_TYPE_BUY || trans.deal_type == DEAL_TYPE_SELL)) {
      if (positionInfo.SelectByTicket(trans.position)) {
         double profit = positionInfo.Profit() + positionInfo.Swap() + positionInfo.Commission();
         LogTradeOutcome(trans.deal, positionInfo.Symbol(), positionInfo.Comment(), profit);
      }
   }
}
