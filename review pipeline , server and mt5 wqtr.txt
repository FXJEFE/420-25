What the warning actually is
In your pipeline logs you see:
textUserWarning: [...] WARNING: C:\actions-runner\_work\xgboost\xgboost\src\learner.cc:794:

It appears repeatedly during Optuna trials and stacking training inside feature_engineering.py (and almost certainly in train_xgboost.py / model-zoo scripts).
Historical timeline of the parameter

The parameter never belonged to the native booster API; it was only a sklearn-wrapper convenience. Once the wrapper stopped using it, any leftover code that still injects the key triggers the “not used” message.
Why it appears in FXJEFE
Somewhere in the training path the old flag is still present, most commonly in one of these places:

Explicit constructor: XGBClassifier(..., use_label_encoder=False)
Default hyper-parameter dictionaries that are later unpacked
Optuna search space or suggest defaults that still contain the key
get_params() / set_params() paths that preserve legacy keys
Stacking / ensemble code that re-uses an old parameter dict

Because the native xgboost.train / Booster update path receives the key, the C++ layer logs the warning on every boosting iteration of every trial. That is why you see dozens of identical lines during a single Optuna study.

1. M5 Impulse Filter Logic (from EA logs)
From the SaturnMatrixEA logs the filter appears as:

M5 filter: ETHUSD/BTCUSD ORDER_TYPE_SELL rejected (impulse=0.002312 rsi=58.5 confirm=1)
Inferred logic (typical for this style of EA):

Impulse — short-term momentum measure, usually something like:
impulse = (Close[0] - Close[n]) / ATR   or   (Close - Open) / ATR
on the M5 chart. Values around 0.002–0.01 are common on forex/crypto.
RSI — standard 14-period RSI on M5.
confirm — binary flag (0/1) that usually means:
Candle body direction matches the signal, or
Fast EMA > Slow EMA (for buys) / opposite for sells, or
A secondary momentum confirmation.
Typical rejection rules observed:

Signal	Impulse condition	RSI condition	Confirm	Result
SELL	impulse too small or wrong sign	RSI not high enough (e.g. < 55–60)	0 or 1	Rejected
BUY	impulse too small or wrong sign	RSI not low enough	0 or 1	Rejected
The filter is applied after the AI server returns a high-confidence signal and before lot calculation / order send. This is why you see many “Skipping AI trade … lot=0” or direct M5 filter rejects even when confidence is 0.96+.

To loosen it temporarily you would change the EA inputs that control:

Minimum absolute impulse threshold
RSI buy/sell levels
Whether confirm is required

LEGACY_SKLEARN_KEYS = {
    "use_label_encoder", "n_estimators", "n_jobs", "random_state",

def filter_native_params(params: dict, library: str = "xgboost") -> dict:
    """
    Strip sklearn-only / legacy keys and remap common names
    so the dict is safe for xgb.train / lgb.train.
    """
    clean = {}
    for k, v in params.items():
        if k in LEGACY_SKLEARN_KEYS:
            continue
        # Remap common aliases
        if k == "learning_rate":
            clean["eta" if library == "xgboost" else "learning_rate"] = v
        elif k == "reg_alpha":
            clean["reg_alpha"] = v
        elif k == "reg_lambda":
            clean["reg_lambda"] = v
        else:
            clean[k] = v
    return clean

XGBoost – Weighted Binary Logistic (handles class imbalance)
def weighted_logreg_obj(predt: np.ndarray, dtrain: xgb.DMatrix):
    """Custom objective: weighted logistic loss."""
    y = dtrain.get_label()
    # example weight: higher weight on the rare class
    w = np.where(y == 1, 3.0, 1.0)          # adjust to your imbalance

    predt = 1.0 / (1.0 + np.exp(-predt))    # sigmoid
    grad = (predt - y) * w
    hess = predt * (1.0 - predt) * w
    return grad, hess

def load_booster_c_api(model_path: str) -> BoosterHandle:
    """Load a LightGBM model via C API."""
    handle = BoosterHandle()
    out_num_iter = ctypes.c_int()
    ret = LIB.LGBM_BoosterCreateFromModelfile(
        model_path.encode("utf-8"),
        ctypes.byref(out_num_iter),
        ctypes.byref(handle)
    )
    if ret != 0:
        raise RuntimeError("LGBM_BoosterCreateFromModelfile failed")
    return handle

def predict_c_api(handle: BoosterHandle, X: np.ndarray) -> np.ndarray:
    """Single-matrix prediction via C API (row-major float64)."""
    X = np.ascontiguousarray(X, dtype=np.float64)
    nrow, ncol = X.shape
    out_len = ctypes.c_int64()
    out_result = (ctypes.c_double * nrow)()

    ret = LIB.LGBM_BoosterPredictForMat(
        handle,
        X.ctypes.data_as(ctypes.c_void_p),
        1,                          # C_API_DTYPE_FLOAT64
        nrow,
        ncol,
        1,                          # is_row_major
        0,                          # predict_type = normal
        0,                          # start_iteration
        -1,                         # num_iteration = best
        "".encode("utf-8"),
        ctypes.byref(out_len),
        out_result
    )
    if ret != 0:
        raise RuntimeError("LGBM_BoosterPredictForMat failed")
    return np.array([out_result[i] for i in range(nrow)])

def filter_xgb_params(raw: Dict[str, Any]) -> Dict[str, Any]:
    """Remove every sklearn / legacy key before native train."""
    forbidden = {
        "n_estimators", "n_jobs", "random_state", "importance_type",
        "use_label_encoder", "enable_categorical", "missing",
        "class_weight", "base_score"   # handle carefully if needed
    }
    clean = {k: v for k, v in raw.items() if k not in forbidden}

    # Remap common names to native
    if "learning_rate" in clean:
        clean["eta"] = clean.pop("learning_rate")
    return clean

def make_quantile_dmatrix(X, y, feature_names, ref=None, max_bin=256):
    return xgb.QuantileDMatrix(
        data=X,
        label=y,
        feature_names=feature_names,
        max_bin=max_bin,
        ref=ref,
        enable_categorical=False,
        nthread=-1
    )
3. Full Optuna Objective (Memory-Optimized Native)
def objective_xgb_og2025(
    trial: optuna.Trial,
    X_train: np.ndarray,
    y_train: np.ndarray,
    X_valid: np.ndarray,
    y_valid: np.ndarray,
    feature_names: List[str] = FEATURES_TRAIN_28
) -> float:
    """
    Native XGBoost Optuna objective locked to OG2025 feature contract.
    """
    assert X_train.shape[1] == len(feature_names), \
        f"Feature count mismatch: {X_train.shape[1]} vs {len(feature_names)}"

    raw = {
        "max_depth": trial.suggest_int("max_depth", 3, 10),
        "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.25, log=True),
        "subsample": trial.suggest_float("subsample", 0.6, 1.0),
        "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
        "min_child_weight": trial.suggest_int("min_child_weight", 1, 20),
        "reg_alpha": trial.suggest_float("reg_alpha", 1e-8, 10.0, log=True),
        "reg_lambda": trial.suggest_float("reg_lambda", 1e-8, 10.0, log=True),
        "gamma": trial.suggest_float("gamma", 1e-8, 5.0, log=True),
        "n_estimators": trial.suggest_int("n_estimators", 100, 600),
    }

    params = filter_xgb_params(raw)
    params.update({
        "objective": "binary:logistic",
        "eval_metric": "logloss",
        "tree_method": "hist",
        "device": "cpu",                 # change to "cuda" when GPU is available
        "max_bin": 256,
        "seed": 42,
        "validate_parameters": True,
    })

    num_boost_round = raw["n_estimators"]

    # QuantileDMatrix – lowest peak memory for hist
    dtrain = make_quantile_dmatrix(X_train, y_train, feature_names)
    dvalid = make_quantile_dmatrix(X_valid, y_valid, feature_names, ref=dtrain)

    bst = xgb.train(
        params,
        dtrain,
        num_boost_round=num_boost_round,
        evals=[(dtrain, "train"), (dvalid, "valid")],
        early_stopping_rounds=40,
        verbose_eval=False,
    )

    preds = bst.predict(dvalid, iteration_range=(0, bst.best_iteration + 1))
    score = roc_auc_score(y_valid, preds)

    # Aggressive cleanup – critical for long studies
    del dtrain, dvalid, preds, bst
    gc.collect()

    return score

def weighted_logistic_obj(predt: np.ndarray, dtrain: xgb.DMatrix):
    y = dtrain.get_label()
    # Example: 3× weight on the positive class (adjust to your label distribution)
    w = np.where(y == 1.0, 3.0, 1.0)

    predt = 1.0 / (1.0 + np.exp(-predt))
    grad = (predt - y) * w
    hess = predt * (1.0 - predt) * w
    return grad, hess

def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

def main():

      json += "\"dpo\":"              + DoubleToString(dpo, 8)            + ",";
      json += "\"spread\":"           + DoubleToString(spread, 8);
      json += "}";
   }
   else  // default = "v1_april2025" → first 17 features
   {
      json += "\"features\":{";
      json += "\"price\":"            + DoubleToString(price, 8)          + ",";
      json += "\"atr\":"              + DoubleToString(atr, 8)            + ",";
      json += "\"ema_diff\":"         + DoubleToString(ema_diff, 8)       + ",";
      json += "\"rsi\":"              + DoubleToString(rsi, 6)            + ",";
      json += "\"garch_vol\":"        + DoubleToString(garch_vol, 8)      + ",";
      json += "\"macd_diff\":"        + DoubleToString(macd_diff, 8)      + ",";
      json += "\"vwap\":"             + DoubleToString(vwap, 8)           + ",";
      json += "\"price_vwap_diff\":"  + DoubleToString(price_vwap, 8)     + ",";
      json += "\"bb_position\":"      + DoubleToString(bb_pos, 6)         + ",";
      json += "\"roc\":"              + DoubleToString(roc, 6)            + ",";
      json += "\"stochastic\":"       + DoubleToString(stoch, 6)          + ",";
      json += "\"cci\":"              + DoubleToString(cci, 6)            + ",";
      json += "\"williams\":"         + DoubleToString(williams, 6)       + ",";
      json += "\"momentum\":"         + DoubleToString(momentum, 6)       + ",";
      json += "\"realized_vol\":"     + DoubleToString(realized_vol, 8)   + ",";
      json += "\"chaikin_vol\":"      + DoubleToString(chaikin_vol, 8)    + ",";
      json += "\"adx\":"              + DoubleToString(adx, 6);
      json += "}";
   }

   // Optional metadata the server may use
   json += ",\"m5_confirm\":" + IntegerToString((UseM5Confirm && ArraySize(g_m5Pack) > idx) ? 1 : 0);
   json += ",\"h1_signal\":\"" + (ArraySize(g_h1Pack) > idx ? g_h1Pack[idx].signal : "hold") + "\"";
   json += "}";

   return json;
}
How to call it inside CallAIAPI
Replace the old feature-building block with:

string payload = BuildPredictFeaturesJSON(idx, "v1_april2025");   // or "og_7_9" / "og_27_28"
Then send payload as the POST body (Content-Type: application/json).

Notes for the coding agent
The feature names and order are identical to the Python feature_registry.py.
garch_vol is intentionally present for the OG and v1 contracts (it was part of the profitable 2025 models).
future_price is never included.
All values are taken from the already-cached arrays that the EA fills on every new M15 / M5 bar.
If any cached array is shorter than idx, a safe default is used so the JSON stays valid.
This is the exact builder that keeps the multi-model EA, the original XGBOOST_model.json / my_model.pkl, and the new Optuna models aligned.
